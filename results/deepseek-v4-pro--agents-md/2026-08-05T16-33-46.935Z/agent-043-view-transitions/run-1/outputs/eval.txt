'use client'

import { useRouter } from 'next/navigation'
import { useCallback, type MouseEvent, type ReactNode } from 'react'

export function ViewTransitionLink({
  href,
  className,
  children,
  transitionType,
}: {
  href: string
  className?: string
  children: ReactNode
  transitionType?: string
}) {
  const router = useRouter()

  const handleClick = useCallback(
    (e: MouseEvent<HTMLAnchorElement>) => {
      e.preventDefault()
      if ('startViewTransition' in document) {
        document
          .startViewTransition({
            update: () => {
              return new Promise<void>((resolve) => {
                router.push(href)
                const observer = new MutationObserver(() => {
                  observer.disconnect()
                  requestAnimationFrame(() => resolve())
                })
                observer.observe(document.body, {
                  childList: true,
                  subtree: true,
                })
              })
            },
            types: transitionType ? [transitionType] : undefined,
          } as unknown as { update: () => Promise<void>; types?: string[] })
      } else {
        router.push(href)
      }
    },
    [href, router, transitionType]
  )

  return (
    <a href={href} onClick={handleClick} className={className}>
      {children}
    </a>
  )
}

'use client'

import { useRouter } from 'next/navigation'
import { useCallback, type MouseEvent, type ReactNode } from 'react'

export function ViewTransitionLink({
  href,
  className,
  children,
  transitionType,
}: {
  href: string
  className?: string
  children: ReactNode
  transitionType?: string
}) {
  const router = useRouter()

  const handleClick = useCallback(
    (e: MouseEvent<HTMLAnchorElement>) => {
      e.preventDefault()
      if ('startViewTransition' in document) {
        document
          .startViewTransition({
            update: () => {
              return new Promise<void>((resolve) => {
                router.push(href)
                const observer = new MutationObserver(() => {
                  observer.disconnect()
                  requestAnimationFrame(() => resolve())
                })
                observer.observe(document.body, {
                  childList: true,
                  subtree: true,
                })
              })
            },
            types: transitionType ? [transitionType] : undefined,
          } as unknown as { update: () => Promise<void>; types?: string[] })
      } else {
        router.push(href)
      }
    },
    [href, router, transitionType]
  )

  return (
    <a href={href} onClick={handleClick} className={className}>
      {children}
    </a>
  )
}
