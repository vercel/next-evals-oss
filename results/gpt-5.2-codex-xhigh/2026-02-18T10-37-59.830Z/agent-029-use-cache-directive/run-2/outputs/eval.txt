const simulateWork = (durationMs: number) =>
  new Promise((resolve) => setTimeout(resolve, durationMs));

const buildSnapshot = (version: number): CatalogSnapshot => {
  const updatedAt = Date.now();
  const products = baseCatalog.map((product, index) => {
    const price = product.basePrice + (version % 4) * 3 + index;
    const inventory = Math.max(
      0,
      product.baseInventory - ((version * 7 + index * 3) % 26)
    );
    const availability: CatalogProduct['availability'] =
      inventory === 0 ? 'Backorder' : inventory < 20 ? 'Low stock' : 'In stock';

    return {
      id: product.id,
      sku: product.sku,
      name: product.name,
      category: product.category,
      vendor: product.vendor,
      price,
      inventory,
      availability,
      status: product.status
    };
  });

  return {
    products,
    version,
    updatedAt
  };
};

export const getCatalog = unstable_cache(

    return catalogSnapshot;

export const enqueueCatalogSync = async () => {
  if (syncInFlight) {
    return;
  }

  const nextVersion = catalogSnapshot.version + 1;
  syncInFlight = (async () => {
    const nextSnapshot = await fetchCatalogFromERP(nextVersion);
    catalogSnapshot = nextSnapshot;
    revalidateTag(PRODUCTS_CACHE_TAG);
  })();

  syncInFlight.finally(() => {
    syncInFlight = null;
  });
};
