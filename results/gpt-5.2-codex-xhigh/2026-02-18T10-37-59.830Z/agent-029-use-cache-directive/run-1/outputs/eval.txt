  const priceMultiplier = index % 2 === 0 ? 1.02 : 0.985;
  const inventoryDelta = (index % 3 === 0 ? -8 : 6) + Math.floor(Math.random() * 6);
  const nextInventory = clamp(product.inventory + inventoryDelta, 0, 180);
  const nextPrice = Math.round(product.price * priceMultiplier * 100) / 100;

  return {
    ...product,
    price: nextPrice,
    inventory: nextInventory,
    availability: getAvailability(nextInventory),
    updatedAt: new Date().toISOString()
  };
};

export async function runCatalogSync() {
  if (syncInFlight) {
    return { skipped: true };

  syncInFlight = true;
  await delay(1200);

  products = products.map(updateProduct);
  lastSyncedAt = new Date();
  syncInFlight = false;

}
