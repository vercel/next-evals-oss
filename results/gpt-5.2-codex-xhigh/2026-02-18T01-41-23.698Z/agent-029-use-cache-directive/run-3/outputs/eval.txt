import { revalidateTag, unstable_cache } from 'next/cache';

import { getAllProducts } from './db';

export const LOW_STOCK_THRESHOLD = 8;

export type Product = {
  id: number;
  sku: string;
  name: string;
  category: string;
  price: number;
  inventory: number;
  updatedAt: string;
};

export type CatalogTotals = {
  totalSkus: number;
  inStock: number;
  lowStock: number;
  outOfStock: number;
  totalUnits: number;
  inventoryValue: number;
};

export type CatalogSnapshot = {
  products: Product[];
  totals: CatalogTotals;
  generatedAt: string;
};

const getCachedCatalog = unstable_cache(
  async (): Promise<CatalogSnapshot> => {
    const products = await getAllProducts();
    const totals = products.reduce<CatalogTotals>(
      (acc, product) => {
        const isOut = product.inventory === 0;
        const isLow = product.inventory > 0 && product.inventory < LOW_STOCK_THRESHOLD;

        acc.totalSkus += 1;
        acc.totalUnits += product.inventory;
        acc.inventoryValue += product.price * product.inventory;
        acc.outOfStock += isOut ? 1 : 0;
        acc.lowStock += isLow ? 1 : 0;
        acc.inStock += !isOut ? 1 : 0;
        return acc;
      },
      {
        totalSkus: 0,
        inStock: 0,
        lowStock: 0,
        outOfStock: 0,
        totalUnits: 0,
        inventoryValue: 0
      }
    );

    return {
      products,
      totals,
      generatedAt: new Date().toISOString()
    };
  },
  ['catalog-products-v1'],
  {
    tags: ['products'],
    revalidate: 900
  }
);

export async function getCatalogSnapshot() {
  return getCachedCatalog();
}

export async function syncCatalogInBackground() {
  await new Promise((resolve) => setTimeout(resolve, 1200));
  revalidateTag('products');
  await getCachedCatalog();
}

import { revalidateTag, unstable_cache } from 'next/cache';

import { getAllProducts } from './db';

export const LOW_STOCK_THRESHOLD = 8;

export type Product = {
  id: number;
  sku: string;
  name: string;
  category: string;
  price: number;
  inventory: number;
  updatedAt: string;
};

export type CatalogTotals = {
  totalSkus: number;
  inStock: number;
  lowStock: number;
  outOfStock: number;
  totalUnits: number;
  inventoryValue: number;
};

export type CatalogSnapshot = {
  products: Product[];
  totals: CatalogTotals;
  generatedAt: string;
};

const getCachedCatalog = unstable_cache(
  async (): Promise<CatalogSnapshot> => {
    const products = await getAllProducts();
    const totals = products.reduce<CatalogTotals>(
      (acc, product) => {
        const isOut = product.inventory === 0;
        const isLow = product.inventory > 0 && product.inventory < LOW_STOCK_THRESHOLD;

        acc.totalSkus += 1;
        acc.totalUnits += product.inventory;
        acc.inventoryValue += product.price * product.inventory;
        acc.outOfStock += isOut ? 1 : 0;
        acc.lowStock += isLow ? 1 : 0;
        acc.inStock += !isOut ? 1 : 0;
        return acc;
      },
      {
        totalSkus: 0,
        inStock: 0,
        lowStock: 0,
        outOfStock: 0,
        totalUnits: 0,
        inventoryValue: 0
      }
    );

    return {
      products,
      totals,
      generatedAt: new Date().toISOString()
    };
  },
  ['catalog-products-v1'],
  {
    tags: ['products'],
    revalidate: 900
  }
);

export async function getCatalogSnapshot() {
  return getCachedCatalog();
}

export async function syncCatalogInBackground() {
  await new Promise((resolve) => setTimeout(resolve, 1200));
  revalidateTag('products');
  await getCachedCatalog();
}
