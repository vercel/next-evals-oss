async function Project({
  params,
}: Pick<PageProps<'/projects/[slug]'>, 'params'>) {
  const { slug } = await params
  const [title, details] = await Promise.all([
    getProjectTitle(slug),
    getProjectDetails(slug),

    <main>
      <h1>{title}</h1>

        ))}
      </ul>
    </main>
  )
}
