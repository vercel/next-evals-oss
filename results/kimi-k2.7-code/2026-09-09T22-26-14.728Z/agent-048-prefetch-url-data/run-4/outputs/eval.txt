import { useEffect, useState } from 'react'
import { prefetchCategory, readCategory, type CategoryData } from '@/lib/cache'

  const searchParams = useSearchParams()
  const category = searchParams.get('category') ?? 'featured'
  const [data, setData] = useState<CategoryData | undefined>(() =>
    readCategory(category)
  )

  useEffect(() => {

          ))}
        </ul>
      </section>
    </main>
  )
}

      </ul>

  )
}

---FILE---
import { getProducts } from './data'

export type CategoryData = {
  heading: string
  products: string[]
}

const categoryHeading: Record<string, string> = {
  featured: 'Featured products',
  sale: 'Products on sale',
}

const cache = new Map<string, CategoryData>()

export async function prefetchCategory(slug: string) {
  if (cache.has(slug)) return

  const products = await getProducts(slug)
  const heading = categoryHeading[slug] ?? 'Products'

  cache.set(slug, { heading, products })
}

export function readCategory(slug: string): CategoryData | undefined {
  return cache.get(slug)
}

---FILE---
const productsByCategory: Record<string, string[]> = {
  featured: ['Desk Lamp', 'Wool Blanket'],
  sale: ['Canvas Tote', 'Travel Mug'],
}

export async function getProducts(category: string) {
  await new Promise((resolve) => setTimeout(resolve, 800))
  return productsByCategory[category] ?? []
}
