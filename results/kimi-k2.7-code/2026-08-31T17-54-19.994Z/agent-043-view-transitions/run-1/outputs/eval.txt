'use client'

import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useRef,
  useState,
  type ReactNode,
} from 'react'
import { usePathname, useRouter } from 'next/navigation'

type Direction = 'forward' | 'backward'

interface TransitionContextValue {
  direction: Direction | null
  navigate: (href: string, direction?: Direction) => void
}

const TransitionContext = createContext<TransitionContextValue | null>(null)

function getDirection(from: string, to: string): Direction | null {
  if (from === '/' && to.startsWith('/product/')) return 'forward'
  if (from.startsWith('/product/') && to === '/') return 'backward'
  return null
}

export function useTransitionNavigation() {
  const ctx = useContext(TransitionContext)
  if (!ctx) {
    throw new Error(
      'useTransitionNavigation must be used within a TransitionProvider'
    )
  }
  return ctx
}

export function TransitionProvider({ children }: { children: ReactNode }) {
  const router = useRouter()
  const pathname = usePathname()
  const previousPathRef = useRef(pathname)
  const [direction, setDirection] = useState<Direction | null>(null)
  const [reducedMotion, setReducedMotion] = useState(false)

  useEffect(() => {
    const mq = window.matchMedia('(prefers-reduced-motion: reduce)')
    setReducedMotion(mq.matches)
    const onChange = (e: MediaQueryListEvent) => setReducedMotion(e.matches)
    mq.addEventListener('change', onChange)
    return () => mq.removeEventListener('change', onChange)
  }, [])

  useEffect(() => {
    previousPathRef.current = pathname
  }, [pathname])

  const navigate = useCallback(
    (href: string, explicitDirection?: Direction) => {
      const dir = explicitDirection ?? getDirection(previousPathRef.current, href)
      setDirection(dir)

      if (
        reducedMotion ||
        typeof document === 'undefined' ||
        !('startViewTransition' in document)
      ) {
        router.push(href)
        return
      }

      document.documentElement.setAttribute(
        'data-transition-direction',
        dir ?? ''
      )
      document.startViewTransition(() => router.push(href))
    },
    [reducedMotion, router]
  )

  useEffect(() => {
    const onPopState = () => {
      const to = window.location.pathname
      const dir = getDirection(previousPathRef.current, to)
      setDirection(dir)
      document.documentElement.setAttribute(
        'data-transition-direction',
        dir ?? ''
      )
    }

    window.addEventListener('popstate', onPopState)
    return () => window.removeEventListener('popstate', onPopState)
  }, [])

  return (
    <TransitionContext.Provider value={{ direction, navigate }}>
      {children}
    </TransitionContext.Provider>
  )
}

import { TransitionLink } from '@/app/TransitionLink'
import { getProduct, products } from '@/lib/products'
import { notFound } from 'next/navigation'

async function ProductDetails({ slug }: { slug: string }) {
  await new Promise((resolve) => setTimeout(resolve, 100))
  const product = getProduct(slug)
  if (!product) notFound()

  return (
    <div className=\"product-info fade-in\">
      <h1>{product.name}</h1>
      <p className=\"product-price\">${product.price}</p>
      <p className=\"product-description\">{product.description}</p>
    </div>
  )
}

function DetailsSkeleton() {
  return (
    <div className=\"product-info\">
      <div className=\"skeleton-text large\" />
      <div className=\"skeleton-text short\" />
      <div className=\"skeleton-text\" />
    </div>
  )
}

export default async function ProductPage({
  params,
}: {
  params: Promise<{ slug: string }>
}) {
  const { slug } = await params
  const product = products.find((p) => p.slug === slug)
  if (!product) notFound()

  return (
    <div>
      <TransitionLink
        href=\"/\"
        direction=\"backward\"
        className=\"back-link\"
      >
        ← Back to products
      </TransitionLink>
      <div className=\"product-detail\">
        <div
          className=\"product-hero\"
          style={{
            backgroundColor: product.color,
            viewTransitionName: `product-image-${product.slug}`,
          }}
        />
        <Suspense fallback={<DetailsSkeleton />}>
          <ProductDetails slug={slug} />
        </Suspense>
      </div>
    </div>
  )
}

'use client'

import Link from 'next/link'
import { useTransitionNavigation } from './TransitionProvider'
import type { ComponentPropsWithoutRef } from 'react'

type TransitionLinkProps = ComponentPropsWithoutRef<typeof Link> & {
  direction?: 'forward' | 'backward'
}

export function TransitionLink({
  href,
  direction,
  onClick,
  ...props
}: TransitionLinkProps) {
  const { navigate } = useTransitionNavigation()

  return (
    <Link
      href={href}
      onClick={(e) => {
        e.preventDefault()
        onClick?.(e)
        navigate(typeof href === 'string' ? href : href.toString(), direction)
      }}
      {...props}
    />
  )
}

'use client'

import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useRef,
  useState,
  type ReactNode,
} from 'react'
import { usePathname, useRouter } from 'next/navigation'

type Direction = 'forward' | 'backward'

interface TransitionContextValue {
  direction: Direction | null
  navigate: (href: string, direction?: Direction) => void
}

const TransitionContext = createContext<TransitionContextValue | null>(null)

function getDirection(from: string, to: string): Direction | null {
  if (from === '/' && to.startsWith('/product/')) return 'forward'
  if (from.startsWith('/product/') && to === '/') return 'backward'
  return null
}

export function useTransitionNavigation() {
  const ctx = useContext(TransitionContext)
  if (!ctx) {
    throw new Error(
      'useTransitionNavigation must be used within a TransitionProvider'
    )
  }
  return ctx
}

export function TransitionProvider({ children }: { children: ReactNode }) {
  const router = useRouter()
  const pathname = usePathname()
  const previousPathRef = useRef(pathname)
  const [direction, setDirection] = useState<Direction | null>(null)
  const [reducedMotion, setReducedMotion] = useState(false)

  useEffect(() => {
    const mq = window.matchMedia('(prefers-reduced-motion: reduce)')
    setReducedMotion(mq.matches)
    const onChange = (e: MediaQueryListEvent) => setReducedMotion(e.matches)
    mq.addEventListener('change', onChange)
    return () => mq.removeEventListener('change', onChange)
  }, [])

  useEffect(() => {
    previousPathRef.current = pathname
  }, [pathname])

  const navigate = useCallback(
    (href: string, explicitDirection?: Direction) => {
      const dir = explicitDirection ?? getDirection(previousPathRef.current, href)
      setDirection(dir)

      if (
        reducedMotion ||
        typeof document === 'undefined' ||
        !('startViewTransition' in document)
      ) {
        router.push(href)
        return
      }

      document.documentElement.setAttribute(
        'data-transition-direction',
        dir ?? ''
      )
      document.startViewTransition(() => router.push(href))
    },
    [reducedMotion, router]
  )

  useEffect(() => {
    const onPopState = () => {
      const to = window.location.pathname
      const dir = getDirection(previousPathRef.current, to)
      setDirection(dir)
      document.documentElement.setAttribute(
        'data-transition-direction',
        dir ?? ''
      )
    }

    window.addEventListener('popstate', onPopState)
    return () => window.removeEventListener('popstate', onPopState)
  }, [])

  return (
    <TransitionContext.Provider value={{ direction, navigate }}>
      {children}
    </TransitionContext.Provider>
  )
}
