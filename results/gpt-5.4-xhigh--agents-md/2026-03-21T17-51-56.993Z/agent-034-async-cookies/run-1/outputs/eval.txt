import { cookies, headers } from 'next/headers'

function getPreferredLanguage(acceptLanguage: string | null): string {
  if (!acceptLanguage) {
    return 'Not specified'
  }

  const preferredLanguage = acceptLanguage
    .split(',')
    .map((entry, index) => {
      const [languageTag, ...parameters] = entry.trim().split(';')
      const qualityValue = parameters
        .map((parameter) => parameter.trim())
        .find((parameter) => parameter.startsWith('q='))

      const quality = qualityValue ? Number(qualityValue.slice(2)) : 1

      return {
        index,
        languageTag,
        quality: Number.isFinite(quality) ? quality : 0,
      }
    })
    .filter(({ languageTag, quality }) => languageTag.length > 0 && quality > 0)
    .sort(
      (first, second) =>
        second.quality - first.quality || first.index - second.index,
    )[0]

  return preferredLanguage?.languageTag ?? 'Not specified'
}

export default async function UserPreferences() {
  const [cookieStore, headerStore] = await Promise.all([cookies(), headers()])
  const theme = cookieStore.get('theme')?.value ?? 'light'
