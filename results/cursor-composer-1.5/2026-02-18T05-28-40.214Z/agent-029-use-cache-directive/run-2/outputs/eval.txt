 */
async function fetchFromErp(): Promise<Product[]> {
  // Simulate network latency and processing (e.g. thousands of SKUs)
  await new Promise((r) => setTimeout(r, 50));

  const current = getCatalog();
  // Simulate upstream changes: slight price/inventory tweaks
  const now = Date.now();
  return current.map((p, i) => {
    const newInventory = Math.max(0, p.inventory + ((now + i) % 7) - 3);
    return {
      ...p,
      price: Math.round((p.price * (1 + (i % 3) * 0.01)) * 100) / 100,
      inventory: newInventory,
      available: newInventory > 0,
    };
  });
}

  const products = await fetchFromErp();
  setCatalog(products);
  revalidateTag(\"products\");
}
