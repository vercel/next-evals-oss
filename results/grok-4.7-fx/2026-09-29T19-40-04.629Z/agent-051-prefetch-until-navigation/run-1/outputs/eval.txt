'use client'

import { useSearchParams } from 'next/navigation'
import { Suspense, use } from 'react'

export function ProjectHeading({ title }: { title: Promise<string> }) {
  return (
    <Suspense fallback={<h1>Loading project...</h1>}>
      <Heading title={title} />
    </Suspense>
  )
}

function Heading({ title }: { title: Promise<string> }) {
  const linkedTitle = useSearchParams().get('title')

  return (
    <h1>
      <Suspense fallback={linkedTitle || 'Loading project...'}>
        <ResolvedTitle title={title} linkedTitle={linkedTitle} />
      </Suspense>
    </h1>
  )
}

function ResolvedTitle({
  title,
  linkedTitle,
}: {
  title: Promise<string>
  linkedTitle: string | null
}) {
  return use(title) || linkedTitle || 'Unknown project'
}

const projects: Record<
  string,
  { title: string; description: string; activity: string[] }
> = {
  acme: {
    title: 'Acme Storefront',
    description: 'Commerce storefront and checkout.',
    activity: ['Production deployment', 'Checkout updated', 'Domain verified'],
  },
  orbit: {
    title: 'Orbit Analytics',
    description: 'Product analytics and reporting.',
    activity: ['Report published', 'Data source added', 'Team invited'],
  },
}

export async function getProjectTitle(slug: string) {
  'use cache'
  return projects[slug]?.title ?? 'Unknown project'
}

export async function getProjectDetails(slug: string) {
  await new Promise((resolve) => setTimeout(resolve, 800))
  return (
    projects[slug] ?? {
      title: 'Unknown project',
      description: 'This project could not be found.',
      activity: [],
    }
  )
}
