let syncVersion = 0;
let syncInFlight = false;

function wait(ms) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

function toAvailability(inventory) {
  if (inventory <= 0) {
    return \"out_of_stock\";
  }

  if (inventory < 15) {
    return \"low_stock\";
  }

  return \"in_stock\";
}

export async function getAllProducts() {
  await wait(WAIT_FOR_DB_MS);
  return products.map((product) => ({ ...product }));
}

export async function syncLatestCatalogFromUpstream() {
  if (syncInFlight) {
    return;
  }

  syncInFlight = true;

  try {
    await wait(WAIT_FOR_SYNC_MS);

    syncVersion += 1;
    const updatedAt = new Date().toISOString();

    products = products.map((product, index) => {
      const priceBump = 120 * ((syncVersion + index) % 4);
      const inventoryShift = ((index % 5) - 2) * syncVersion;
      const nextInventory = Math.max(0, product.inventory + inventoryShift);

      return {
        ...product,
        priceCents: Math.max(500, product.priceCents + priceBump),
        inventory: nextInventory,
        availability: toAvailability(nextInventory),
        updatedAt
      };
    });
  } finally {
    syncInFlight = false;
  }
}
