import { Suspense } from 'react'
import { getProjectDetails, getProjectTitle } from '@/lib/data'

export default function ProjectPage({
  params,
  searchParams,
}: PageProps<'/projects/[slug]'>) {
  return (
    <main>
      <Suspense fallback={<h1>Loading project...</h1>}>
        <ProjectTitle params={params} searchParams={searchParams} />
      </Suspense>
      <Suspense fallback={<p>Loading details...</p>}>
        <ProjectDetails params={params} />
      </Suspense>
    </main>
  )
}

async function ProjectTitle({
  params,
  searchParams,
}: Pick<PageProps<'/projects/[slug]'>, 'params' | 'searchParams'>) {
  const { title: initialTitle } = await searchParams
  if (initialTitle) return <h1>{initialTitle}</h1>

  const { slug } = await params
  const title = await getProjectTitle(slug)
  return <h1>{title}</h1>
}

async function ProjectDetails({
  params,
}: Pick<PageProps<'/projects/[slug]'>, 'params'>) {
  const { slug } = await params
  const details = await getProjectDetails(slug)

  return (
    <>
      <p>{details.description}</p>
      <h2>Recent activity</h2>
      <ul>
        {details.activity.map((item) => (
          <li key={item}>{item}</li>
        ))}
      </ul>
    </>
  )
}

const projects: Record<
  string,
  { title: string; description: string; activity: string[] }
> = {
  acme: {
    title: 'Acme Storefront',
    description: 'Commerce storefront and checkout.',
    activity: ['Production deployment', 'Checkout updated', 'Domain verified'],
  },
  orbit: {
    title: 'Orbit Analytics',
    description: 'Product analytics and reporting.',
    activity: ['Report published', 'Data source added', 'Team invited'],
  },
}

export async function getProjectsList() {
  'use cache'
  return Object.entries(projects).map(([slug, { title }]) => ({
    slug,
    title,
  }))
}

export async function getProjectTitle(slug: string) {
  'use cache'
  return projects[slug]?.title ?? 'Unknown project'
}

export async function getProjectDetails(slug: string) {
  'use cache'
  await new Promise((resolve) => setTimeout(resolve, 800))
  return (
    projects[slug] ?? {
      title: 'Unknown project',
      description: 'This project could not be found.',
      activity: [],
    }
  )
}
