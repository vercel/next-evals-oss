import { Suspense } from 'react'
import { getProjectDetails, getProjectTitle } from '@/lib/data'
import { ProjectTitle } from './project-title'

export default function ProjectPage({ params }: PageProps<'/projects/[slug]'>) {
  return (
    <main>
      <ProjectTitle>
        <ProjectTitleServer params={params} />
      </ProjectTitle>
      <Suspense fallback={<p>Loading details...</p>}>
        <ProjectDetails params={params} />
      </Suspense>
    </main>
  )
}

async function ProjectTitleServer({
  params,
}: Pick<PageProps<'/projects/[slug]'>, 'params'>) {
  const { slug } = await params
  const title = await getProjectTitle(slug)

  return <h1>{title}</h1>
}

async function ProjectDetails({
  params,
}: Pick<PageProps<'/projects/[slug]'>, 'params'>) {
  const { slug } = await params
  const details = await getProjectDetails(slug)

  return (
    <>
      <p>{details.description}</p>
      <h2>Recent activity</h2>
      <ul>
        {details.activity.map((item) => (
          <li key={item}>{item}</li>
        ))}
      </ul>
    </>
  )
}

'use client'

import { Suspense, type ReactNode } from 'react'
import { useSearchParams } from 'next/navigation'

function ProjectTitleInner({
  initialTitle,
  children,
}: {
  initialTitle: string | null
  children: ReactNode
}) {
  if (initialTitle) {
    return <h1>{initialTitle}</h1>
  }

  return children
}

function ProjectTitleFromSearchParams({ children }: { children: ReactNode }) {
  const searchParams = useSearchParams()
  const initialTitle = searchParams.get('title')

  return (
    <ProjectTitleInner initialTitle={initialTitle}>{children}</ProjectTitleInner>
  )
}

export function ProjectTitle({ children }: { children: ReactNode }) {
  return (
    <Suspense fallback={<h1>Loading project...</h1>}>
      <ProjectTitleFromSearchParams>{children}</ProjectTitleFromSearchParams>
    </Suspense>
  )
}

export const projectList = [
  { slug: 'acme', name: 'Acme Storefront' },
  { slug: 'orbit', name: 'Orbit Analytics' },
] as const

const projects: Record<
  string,
  { title: string; description: string; activity: string[] }
> = {
  acme: {
    title: 'Acme Storefront',
    description: 'Commerce storefront and checkout.',
    activity: ['Production deployment', 'Checkout updated', 'Domain verified'],
  },
  orbit: {
    title: 'Orbit Analytics',
    description: 'Product analytics and reporting.',
    activity: ['Report published', 'Data source added', 'Team invited'],
  },
}

export async function getProjectTitle(slug: string) {
  'use cache'
  return projects[slug]?.title ?? 'Unknown project'
}

export async function getProjectDetails(slug: string) {
  'use cache'
  await new Promise((resolve) => setTimeout(resolve, 800))
  return (
    projects[slug] ?? {
      title: 'Unknown project',
      description: 'This project could not be found.',
      activity: [],
    }
  )
}
