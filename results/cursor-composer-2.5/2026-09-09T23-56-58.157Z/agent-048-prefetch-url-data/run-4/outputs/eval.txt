'use client'

import { useSearchParams } from 'next/navigation'
import { useEffect, useState } from 'react'
import { loadCategory } from '@/app/actions/category'
import {
  getCachedCategory,
  setCachedCategory,
  type CategoryData,
} from '@/lib/category-cache'

function CategorySection({ heading, products }: CategoryData) {
  return (
    <section>
      <h2>{heading}</h2>
      <ul>
        {products.map((product) => (
          <li key={product}>{product}</li>
        ))}
      </ul>
    </section>
  )
}

function CategoryLoader({ category }: { category: string }) {
  const [data, setData] = useState<CategoryData | null>(null)

  useEffect(() => {
    let cancelled = false

    loadCategory(category).then((result) => {
      if (!cancelled) {
        setCachedCategory(category, result)
        setData(result)
      }
    })

    return () => {
      cancelled = true
    }
  }, [category])

  if (!data) {
    return <p>Loading category...</p>
  }

  return <CategorySection heading={data.heading} products={data.products} />
}

export function CatalogCategoryView() {
  const searchParams = useSearchParams()
  const category = searchParams.get('category') ?? 'featured'
  const cached = getCachedCategory(category)

  if (cached) {
    return <CategorySection heading={cached.heading} products={cached.products} />
  }

  return <CategoryLoader category={category} />
}

import { useEffect } from 'react'
import { loadCategory } from '@/app/actions/category'
import { prefetchCategoryData } from '@/lib/category-cache'

type CategoryLinkProps = {
  slug: string
  label: string
}

function prefetchCategory(slug: string, href: string, router: ReturnType<typeof useRouter>) {
  router.prefetch(href)
  prefetchCategoryData(slug, () => loadCategory(slug))
}

export function CategoryLink({ slug, label }: CategoryLinkProps) {
  const href = `/catalog?category=${slug}`
  const router = useRouter()

  useEffect(() => {
    prefetchCategory(slug, href, router)
  }, [slug, href, router])

  return (
    <Link
      href={href}
      onMouseEnter={() => prefetchCategory(slug, href, router)}
      onFocus={() => prefetchCategory(slug, href, router)}
    >
      {label}
    </Link>
  )
}

export type CategoryData = {
  heading: string
  products: string[]
}

const cache = new Map<string, CategoryData>()
const inflight = new Map<string, Promise<CategoryData>>()

export function getCachedCategory(slug: string) {
  return cache.get(slug)
}

export function setCachedCategory(slug: string, data: CategoryData) {
  cache.set(slug, data)
}

export function prefetchCategoryData(
  slug: string,
  loader: () => Promise<CategoryData>,
) {
  if (cache.has(slug)) {
    return
  }

  if (!inflight.has(slug)) {
    inflight.set(
      slug,
      loader().then((data) => {
        cache.set(slug, data)
        inflight.delete(slug)
        return data
      }),
    )
  }
}

const productsByCategory: Record<string, string[]> = {
  featured: ['Desk Lamp', 'Wool Blanket'],
  sale: ['Canvas Tote', 'Travel Mug'],

export async function getProducts(category: string) {
  'use cache'
  cacheLife('minutes')

  await new Promise((resolve) => setTimeout(resolve, 800))
  return productsByCategory[category] ?? []
}
