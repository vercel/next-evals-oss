'use client'

import { useSearchParams } from 'next/navigation'
import { useEffect, useState } from 'react'
import { CategoryResultsView } from '@/components/category-results-view'
import {
  getCategorySnapshot,
  setCategorySnapshot,
  type CategorySnapshot,
} from '@/lib/category-cache'
import { getCategoryHeading } from '@/lib/categories'

export function CategoryResults() {
  const searchParams = useSearchParams()
  const category = searchParams.get('category') ?? 'featured'
  const cached = getCategorySnapshot(category)
  const [snapshot, setSnapshot] = useState<CategorySnapshot | null>(null)

  useEffect(() => {
    const prepared = getCategorySnapshot(category)
    if (prepared) {
      setSnapshot(prepared)
      return
    }

    let cancelled = false

    void fetch(`/api/catalog?category=${encodeURIComponent(category)}`)
      .then((response) => response.json())
      .then((data: CategorySnapshot) => {
        if (cancelled) {
          return
        }

        setCategorySnapshot(category, data)
        setSnapshot(data)
      })
      .catch(() => {
        if (cancelled) {
          return
        }

        const fallback: CategorySnapshot = {
          heading: getCategoryHeading(category),
          products: [],
        }
        setCategorySnapshot(category, fallback)
        setSnapshot(fallback)
      })

    return () => {
      cancelled = true
    }
  }, [category])

  const display = cached ?? snapshot

  if (!display) {
    return <p>Loading category...</p>
  }

  return <CategoryResultsView heading={display.heading} products={display.products} />
}

import {
  clearInflightPrefetch,
  getCategorySnapshot,
  getInflightPrefetch,
  setCategorySnapshot,
  setInflightPrefetch,
  type CategorySnapshot,
} from '@/lib/category-cache'
import { getCategoryHeading } from '@/lib/categories'

async function fetchCategorySnapshot(category: string): Promise<CategorySnapshot> {
  const response = await fetch(`/api/catalog?category=${encodeURIComponent(category)}`)

  if (!response.ok) {
    throw new Error(`Failed to prefetch category: ${category}`)
  }

  const data = (await response.json()) as CategorySnapshot
  setCategorySnapshot(category, data)
  return data
}

export function prefetchCategory(category: string) {
  const cached = getCategorySnapshot(category)
  if (cached) {
    return Promise.resolve(cached)
  }

  const inflight = getInflightPrefetch(category)
  if (inflight) {
    return inflight
  }

  const promise = fetchCategorySnapshot(category)
    .catch((error) => {
      const fallback: CategorySnapshot = {
        heading: getCategoryHeading(category),
        products: [],
      }
      setCategorySnapshot(category, fallback)
      return fallback
    })
    .finally(() => {
      clearInflightPrefetch(category)
    })

  setInflightPrefetch(category, promise)
  return promise
}
