  const router = useRouter()

  slug: string

}

  { slug: 'featured', label: 'Featured products', heading: 'Featured products' },
  { slug: 'sale', label: 'Products on sale', heading: 'Products on sale' },
]

export function getCategoryHeading(slug: string): string {
  return categories.find((category) => category.slug === slug)?.heading ?? 'Featured products'
}

export function catalogUrl(slug: string): string {
  return `/catalog?category=${slug}`
}

'use client'

import { getCategoryBundle, type CategoryBundle } from '@/app/actions'

const cache = new Map<string, Promise<CategoryBundle>>()

export function prefetchCategory(category: string): void {
  if (!cache.has(category)) {
    cache.set(category, getCategoryBundle(category))
  }
}

export function getCategoryData(category: string): Promise<CategoryBundle> {
  const cached = cache.get(category)
  if (cached) {
    return cached
  }

  const promise = getCategoryBundle(category)
  cache.set(category, promise)
  return promise
}
