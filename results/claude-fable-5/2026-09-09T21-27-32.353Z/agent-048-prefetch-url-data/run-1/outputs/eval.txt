---FILE---
import { Suspense } from 'react'
import { getProducts } from '@/lib/data'
import { categories } from '@/lib/categories'
import { CategoryResults } from './category-results'

export default async function CatalogPage() {
  
  
  const productsByCategory = Object.fromEntries(
    await Promise.all(
      categories.map(
        async ({ slug }) => [slug, await getProducts(slug)] as const
      )
    )
  )

  return (
    <main>
      <h1>Catalog</h1>
      <Suspense fallback={<p>Loading category...</p>}>
        <CategoryResults productsByCategory={productsByCategory} />
      </Suspense>
    </main>
  )
}
