import { revalidateTag } from 'next/cache'
import { getProducts } from '@/lib/products'

export default async function ProductsPage() {
  const products = await getProducts()

  async function syncLatestCatalog() {
    'use server'

    // In production this would enqueue/trigger the ERP/PIM sync job that
    // updates the product store. The sync touches thousands of SKUs, so we
    // must not make the admin's request wait on recomputed catalog data.

    // Stale-while-revalidate invalidation: marks every \"products\" cache entry
    // stale and refreshes it in the background. The admin gets a response
    // immediately (possibly with briefly stale data), and all views reading
    // through the \"products\" tag converge shortly after. updateTag() would
    // instead block this request until the catalog is recomputed.
    revalidateTag('products', 'max')
  }

  return (
    <main>
      <h1>Product Catalog</h1>

      <form action={syncLatestCatalog}>
        <button type=\"submit\">Sync latest catalog</button>
      </form>

      <table>
        <thead>
          <tr>
            <th>ID</th>
            <th>Name</th>
            <th>Price</th>
          </tr>
        </thead>
        <tbody>
          {products.map((product) => (
            <tr key={product.id}>
              <td>{product.id}</td>
              <td>{product.name}</td>
              <td>${product.price}</td>
            </tr>
          ))}
        </tbody>
      </table>
    </main>
  )
}

import { getAllProducts } from './db'
