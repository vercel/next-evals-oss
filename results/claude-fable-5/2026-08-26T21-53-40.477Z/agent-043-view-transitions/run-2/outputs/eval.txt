'use client'

import Link from 'next/link'
import { useRouter } from 'next/navigation'
import { addTransitionType, startTransition, type ComponentProps } from 'react'

type TransitionLinkProps = Omit<ComponentProps<typeof Link>, 'href'> & {
  href: string
  
  direction?: 'forward' | 'back'
}

export function TransitionLink({
  href,
  direction = 'forward',
  onClick,
  ...props
}: TransitionLinkProps) {
  const router = useRouter()

  return (
    <Link
      href={href}
      onClick={(event) => {
        onClick?.(event)
        
        if (
          event.defaultPrevented ||
          event.button !== 0 ||
          event.metaKey ||
          event.ctrlKey ||
          event.shiftKey ||
          event.altKey
        ) {
          return
        }
        event.preventDefault()
        startTransition(() => {
          addTransitionType(
            direction === 'back' ? 'navigation-back' : 'navigation-forward'
          )
          router.push(href)
        })
      }}
      {...props}
    />
  )
}

import { Suspense } from 'react'
import { ViewTransition } from 'react'
import { getProduct, products } from '@/lib/products'
import { notFound } from 'next/navigation'
import { TransitionLink } from '@/app/TransitionLink'

async function ProductDetails({ slug }: { slug: string }) {
  await new Promise((resolve) => setTimeout(resolve, 100))
  const product = getProduct(slug)
  if (!product) notFound()

  return (
    <div className=\"product-info\">
      <h1>{product.name}</h1>
      <p className=\"product-price\">${product.price}</p>
      <p className=\"product-description\">{product.description}</p>
    </div>
  )
}

  )
}

  params,
}: {
  params: Promise<{ slug: string }>
}) {
  const { slug } = await params

  if (!product) notFound()

  return (
    <div>
      <TransitionLink href=\"/\" direction=\"back\" className=\"back-link\">
        ← Back to products

        >
          <ViewTransition enter=\"suspense-fade\">
            <ProductDetails slug={slug} />
          </ViewTransition>
        </Suspense>
      </div>
    </div>
  )
}
