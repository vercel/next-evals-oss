/**
 * Simulates a large ERP/PIM pull (many SKUs). Mutates the in-memory source so
 * the next cache fill reflects upstream changes.
 */
export async function syncLatestFromErp(): Promise<void> {
  await new Promise((resolve) => setTimeout(resolve, 1800))
  const jitter = () => Math.floor(Math.random() * 200) - 100
  pimSnapshot = pimSnapshot.map((p) => {
    const deltaStock = (p.id % 4) - 1
    const nextStock = Math.max(0, p.stock + deltaStock + jitter())
    return {
      ...p,
      priceCents: Math.max(99, p.priceCents + jitter()),
      stock: nextStock,
      available: nextStock > 0,
    }
  })
  lastSyncedAt = new Date().toISOString()
}
