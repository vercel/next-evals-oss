  id: number

  products: Product[]

  generation: number

const BASE_PRODUCTS: Product[] = [
  { id: 1, sku: 'ELEC-LAP-01', name: 'Laptop', price: 999 },
  { id: 2, sku: 'ELEC-PHN-02', name: 'Phone', price: 699 },
  { id: 3, sku: 'ELEC-TAB-03', name: 'Tablet', price: 499 },
  { id: 4, sku: 'ACC-HDP-04', name: 'Headphones', price: 199 },
  { id: 5, sku: 'ACC-WCH-05', name: 'Watch', price: 299 },
]

let catalogGeneration = 0
let lastSyncCompletedAt: string | null = null

/**
 * Simulated ERP/PIM pull — in production this would batch thousands of SKUs
 * and write through to your database or search index.
 */
export async function syncCatalogFromUpstream(): Promise<void> {
  await new Promise((r) => setTimeout(r, 1200))
  catalogGeneration += 1
  lastSyncCompletedAt = new Date().toISOString()
}

async function loadCatalogFromSource(): Promise<CatalogPayload> {
  await new Promise((r) => setTimeout(r, 100))
  return {
    products: BASE_PRODUCTS.map((p) => ({
      ...p,
      price: p.price + catalogGeneration,
    })),
    generation: catalogGeneration,
    lastSyncCompletedAt,
  }
}
