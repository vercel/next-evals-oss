export type CatalogProduct = {
  sku: string
  name: string
  priceCents: number
  stock: number
  available: boolean
}

}

/** Simulated upstream ERP/PIM payload (stable IDs, changing numbers on each pull). */
function fetchUpstreamProductRows(revision: number): CatalogProduct[] {
  const base = [
    { sku: 'SKU-1001', name: 'Laptop 14\"', seed: 999_00 },
    { sku: 'SKU-1002', name: 'Phone', seed: 699_00 },
    { sku: 'SKU-1003', name: 'Tablet', seed: 499_00 },
    { sku: 'SKU-1004', name: 'Headphones', seed: 199_00 },
    { sku: 'SKU-1005', name: 'Watch', seed: 299_00 },
  ]

  const rows: CatalogProduct[] = []
  // Stretch to many SKUs without hand-maintaining a giant list.
  for (let i = 0; i < 1200; i++) {
    const b = base[i % base.length]
    const variant = Math.floor(i / base.length)
    const sku = `${b.sku}-${variant.toString().padStart(3, '0')}`
    const jitter = ((revision * 17 + i * 31) % 97) - 48
    const priceCents = Math.max(99_00, b.seed + jitter * 100)
    const stock = Math.max(0, 200 + jitter - (revision % 7) * 3 - (i % 11))
    rows.push({
      sku,
      name: `${b.name} (${variant})`,
      priceCents,
      stock,
      available: stock > 0,
    })
  }
  return rows
}

let liveSnapshot: CatalogSnapshot = {
  revision: 1,
  lastSyncedAt: new Date().toISOString(),
  products: fetchUpstreamProductRows(1),
}

/**
 * Applies a full catalog refresh from the simulated ERP/PIM.
 * Intentionally slow: represents batch pricing/inventory reconciliation.
 */
export async function materializeCatalogFromUpstream(): Promise<CatalogSnapshot> {
  await new Promise((r) => setTimeout(r, 2200))
  const nextRevision = liveSnapshot.revision + 1
  liveSnapshot = {
    revision: nextRevision,
    lastSyncedAt: new Date().toISOString(),
    products: fetchUpstreamProductRows(nextRevision),
  }
  return liveSnapshot
}

async function readLiveSnapshot(): Promise<CatalogSnapshot> {
  return { ...liveSnapshot, products: liveSnapshot.products }
}

const getCachedCatalog = unstable_cache(

  ['admin-catalog'],
