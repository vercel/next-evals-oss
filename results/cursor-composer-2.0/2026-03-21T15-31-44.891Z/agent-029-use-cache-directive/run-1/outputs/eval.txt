/**
 * Shared catalog read for any route. Tagged so `revalidateTag(\"products\")` refreshes all consumers together.
 */

  async (): Promise<CatalogSnapshot> => {

  /** Increments on each successful pull from ERP */
  generation: 0,
}

/**
 * Full catalog snapshot (read path). Call through `getCachedCatalog` so requests stay on the cache.
 */
export async function getCatalogSnapshot() {
  await delay(100)
  return {
    syncedAt: erpState.lastSyncedAt,
    generation: erpState.generation,
    products: erpState.products.map((p) => ({ ...p })),
  }
}

/**
 * Long-running upstream sync (thousands of SKUs). Run inside `after()` so the admin UI returns immediately.
 */
export async function syncCatalogFromErp() {
  await delay(2200)
  erpState.generation += 1
  erpState.lastSyncedAt = new Date().toISOString()
  erpState.products = erpState.products.map((p) => {
    const delta = erpState.generation % 2 === 0 ? 2 : -1
    const stock = Math.max(0, p.stock + delta)
    return {
      ...p,
      price: p.price + (erpState.generation % 3 === 0 ? -5 : 5),
      stock,
      available: stock > 0,
    }
  })
}
