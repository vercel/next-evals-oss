function formatUsd(cents: number) {
  return new Intl.NumberFormat('en-US', {
    style: 'currency',
    currency: 'USD',
  }).format(cents / 100)
}

export default async function AdminCatalogPage() {
  async function syncLatestCatalog() {
    'use server'
    after(async () => {
      await runCatalogSyncJob()

export type CatalogProduct = {
  sku: string
  name: string
  priceCents: number
  inventory: number
}

type CatalogSnapshot = {
  revision: number
  lastUpstreamSyncAt: string
  products: CatalogProduct[]
}

let revision = 0
let lastUpstreamSyncAt = new Date().toISOString()

const seedNames = [
  'Laptop',
  'Phone',
  'Tablet',
  'Headphones',
  'Watch',
  'Keyboard',
  'Monitor',
  'Webcam',
  'Dock',
  'Router',
]

function buildInitialProducts(): CatalogProduct[] {
  const out: CatalogProduct[] = []
  for (let i = 0; i < 48; i++) {
    const name = seedNames[i % seedNames.length]
    const variant = Math.floor(i / seedNames.length)
    out.push({
      sku: `SKU-${10000 + i}`,
      name: variant > 0 ? `${name} (${variant + 1})` : name,
      priceCents: 9999 + i * 137,
      inventory: 12 + (i * 7) % 180,
    })
  }
  return out
}

let products = buildInitialProducts()

/** Snapshot read by the cached catalog loader (must stay pure aside from reading module state). */
export function readCatalogSnapshot(): CatalogSnapshot {
  return {
    revision,
    lastUpstreamSyncAt,
    products: products.map((p) => ({ ...p })),
  }
}

/**
 * Simulates a long-running ERP/PIM pull (many SKUs). Call from `after()` so the admin
 * request returns immediately; then invalidate the `products` tag so readers pick up fresh data.
 */
export async function runCatalogSyncJob(): Promise<void> {
  await new Promise((r) => setTimeout(r, 1400))
  revision += 1
  products = products.map((p) => ({
    ...p,
    inventory: Math.max(0, p.inventory + Math.floor(Math.random() * 9) - 2),
    priceCents: p.priceCents + (Math.random() > 0.65 ? Math.floor(Math.random() * 5) * 100 : 0),
  }))
  lastUpstreamSyncAt = new Date().toISOString()
}

import { unstable_cache } from 'next/cache'
import { readCatalogSnapshot } from '@/lib/catalog-store'

const getCachedCatalogSnapshot = unstable_cache(
  async () => readCatalogSnapshot(),
  ['admin-catalog-snapshot'],
  { tags: ['products'] }
)

/** Shared catalog read for any admin view; avoids recomputing on every request. */
export function getCachedCatalog() {
  return getCachedCatalogSnapshot()
}
