  return `product-image-${slug}`
}

export function ProductImage({
  slug,
  color,
}: {
  slug: string
  color: string
}) {
  const ref = useRef<HTMLDivElement>(null)

  useEffect(() => {
    const el = ref.current
    if (!el) return
    try {
      if (sessionStorage.getItem(LAST_VIEWED_PRODUCT_KEY) === slug) {
        el.style.setProperty(
          'view-transition-name',
          productImageTransitionName(slug)

          el.style.removeProperty('view-transition-name')
        }
      }
    } catch {}
  }, [slug])

  return (
    <div
      ref={ref}
      className=\"product-image\"
      style={{ backgroundColor: color }}
    />
  )
}

'use client'

import Link from 'next/link'
import { useRouter } from 'next/navigation'
import type { ComponentProps, MouseEvent } from 'react'
import {
  startNavigationTransition,
  type NavigationDirection,
} from './view-transition'

type TransitionLinkProps = ComponentProps<typeof Link> & {
  direction?: NavigationDirection
  prepare?: (anchor: HTMLAnchorElement) => void
}

export function TransitionLink({
  direction = 'forward',
  prepare,
  onClick,
  href,
  ...rest
}: TransitionLinkProps) {
  const router = useRouter()

  const handleClick = (e: MouseEvent<HTMLAnchorElement>) => {
    onClick?.(e)
    if (e.defaultPrevented) return
    if (e.button !== 0 || e.metaKey || e.ctrlKey || e.shiftKey || e.altKey)
      return

    e.preventDefault()
    prepare?.(e.currentTarget)
    startNavigationTransition(
      () => router.push(typeof href === 'string' ? href : href.toString()),
      direction
    )
  }

  return <Link {...rest} href={href} onClick={handleClick} />
}

'use client'

export type NavigationDirection = 'forward' | 'back'

let resolveNavigation: (() => void) | null = null

export function navigationCommitted() {
  resolveNavigation?.()
  resolveNavigation = null
}

export function canAnimateNavigation() {
  return (
    typeof document !== 'undefined' &&
    typeof document.startViewTransition === 'function' &&
    !window.matchMedia('(prefers-reduced-motion: reduce)').matches
  )
}

export function startNavigationTransition(
  navigate: () => void,
  direction: NavigationDirection = 'forward'
) {
  if (!canAnimateNavigation()) {
    navigate()
    return
  }

  document.documentElement.dataset.navDirection = direction

  const transition = document.startViewTransition(() => {
    navigate()
    return new Promise<void>((resolve) => {
      resolveNavigation = resolve
      setTimeout(() => {
        if (resolveNavigation === resolve) resolveNavigation = null
        transition.skipTransition()
      }, 600)
    })
  })

  const clearDirection = () => {
    delete document.documentElement.dataset.navDirection
  }
  transition.finished.then(clearDirection, clearDirection)
}

  )

  navigate: () => void,
  direction: NavigationDirection = 'forward'
) {
  if (!canAnimateNavigation()) {
    navigate()
    return

  document.documentElement.dataset.navDirection = direction

    delete document.documentElement.dataset.navDirection
  }

  color: string

      className=\"product-image\"
      style={{ backgroundColor: color }}
    />
  )
