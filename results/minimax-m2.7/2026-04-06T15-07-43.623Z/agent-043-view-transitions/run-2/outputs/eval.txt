'use client'

import { useRouter } from 'next/navigation'
import Link from 'next/link'
import { startTransition } from 'react'

interface ViewTransitionLinkProps {
  href: string
  children: React.ReactNode
  className?: string
  viewTransitionName?: string
  direction?: 'forward' | 'back'
  onClick?: () => void
}

export function ViewTransitionLink({
  href,
  children,
  className,
  viewTransitionName,
  direction = 'forward',
  onClick,
}: ViewTransitionLinkProps) {
  const router = useRouter()

  const handleClick = (e: React.MouseEvent) => {
    onClick?.()
    
    if (!document.startViewTransition) {
      return
    }
    
    e.preventDefault()
    
    document.body.classList.remove('slide-forward', 'slide-back')
    document.body.classList.add(direction === 'forward' ? 'slide-forward' : 'slide-back')
    
    startTransition(() => {
      router.push(href)
    })
  }

  return (
    <Link
      href={href}
      className={className}
      onClick={handleClick}
      style={viewTransitionName ? { viewTransitionName } : undefined}
    >
      {children}
    </Link>
  )
}

'use client'

import { useRouter } from 'next/navigation'
import Link from 'next/link'
import { startTransition } from 'react'

interface ViewTransitionLinkProps {
  href: string
  children: React.ReactNode
  className?: string
  viewTransitionName?: string
  direction?: 'forward' | 'back'
  onClick?: () => void
}

export function ViewTransitionLink({
  href,
  children,
  className,
  viewTransitionName,
  direction = 'forward',
  onClick,
}: ViewTransitionLinkProps) {
  const router = useRouter()

  const handleClick = (e: React.MouseEvent) => {
    onClick?.()
    
    if (!document.startViewTransition) {
      return
    }
    
    e.preventDefault()
    
    document.body.classList.remove('slide-forward', 'slide-back')
    document.body.classList.add(direction === 'forward' ? 'slide-forward' : 'slide-back')
    
    startTransition(() => {
      router.push(href)
    })
  }

  return (
    <Link
      href={href}
      className={className}
      onClick={handleClick}
      style={viewTransitionName ? { viewTransitionName } : undefined}
    >
      {children}
    </Link>
  )
}
