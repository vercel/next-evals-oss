interface ThemeSwitcherProps {

export default function ThemeSwitcher({ currentTheme }: ThemeSwitcherProps) {
  const handleThemeChange = (newTheme: string) => {
    // Set the theme cookie

  );
}

import { NextRequest, NextResponse } from 'next/server';

export async function GET(request: NextRequest) {
  // Get theme from cookie, default to 'light' if not set
  const theme = request.cookies.get('theme')?.value || 'light';
  
  // Get preferred language from Accept-Language header
  const acceptLanguage = request.headers.get('accept-language');
  let language = 'Unknown';
  
  if (acceptLanguage) {
    // Parse Accept-Language header to get the primary language
    // Accept-Language: \"en-US,en;q=0.9,es;q=0.8\"
    const primaryLanguage = acceptLanguage.split(',')[0]?.split(';')[0]?.trim();
    if (primaryLanguage) {
      language = primaryLanguage;
    }
  }

  return NextResponse.json({
    theme,
    language
  });
