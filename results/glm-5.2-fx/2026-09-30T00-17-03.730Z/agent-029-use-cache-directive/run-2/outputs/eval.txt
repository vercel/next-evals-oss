// Inline Server Action for the \"Sync latest catalog\" trigger.
//
// It returns immediately so the admin can keep working -- the page re-renders
// fast with the (still cached, briefly stale) catalog. The slow sync over
// thousands of SKUs is deferred to `after()`, which runs only after the
// response has been sent. Once the sync completes we invalidate the `products`
// cache tag; the next read (here or in any other view sharing the tag)
// repopulates the cache with fresh data, giving eventual consistency across
// catalog views without forcing every request to block on fresh data.
async function syncCatalogAction() {
  'use server'

  after(async () => {
    await syncCatalogFromUpstream()
    // Purge the `products` cache tag now that the sync has written fresh data.
    // The next read -- on this page or any other view sharing the tag --
    // repopulates the cache, so catalog views converge on the fresh data
    // without forcing every request to block on a recompute.
    revalidateTag('products', 'max')
  })
}

export async function getAllProducts() {
  await new Promise((resolve) => setTimeout(resolve, 100))

export function getCatalogMeta() {
  return { version: catalogVersion, lastSyncedAt }

// Pull the latest catalog from the upstream ERP/PIM into the local store.
// Sync jobs touch thousands of SKUs, so this is intentionally slow -- this is
// exactly the work we do NOT want to block admin requests on. It is run in the
// background (via `after()`) by the sync Server Action.
export async function syncCatalogFromUpstream() {
  // Emulate a bulk sync over thousands of SKUs.
  await new Promise((resolve) => setTimeout(resolve, 1500))

  products = products.map((p) => ({
    ...p,
    // Upstream repricing: ~10% price drop.
    price: Math.max(1, Math.round(p.price * 0.9)),
    // Replenished inventory.
    inventory: p.inventory + 50,
    // Re-availability of previously out-of-stock items.
    available: true,
  }))

  catalogVersion += 1
  lastSyncedAt = new Date().toISOString()

  return { version: catalogVersion, lastSyncedAt }
}

//
// The catalog is read-heavy, so we cache the (slow) DB query under the

export const getProducts = unstable_cache(
  async (): Promise<Catalog> => {

  ['products'],
