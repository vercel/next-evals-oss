/**
 * Inline Server Action, triggered by the <form> below as a regular HTML
 * form submit. It returns immediately so the admin can keep working
 * while the list is briefly stale.
 *
 * The actual upstream sync is deferred to `after()`, which runs only
 * after the response has been sent. Once the (slow, thousands-of-SKUs)
 * sync completes, we invalidate the \"products\" cache tag so the next
 * read — in this view or any other view that depends on the same
 * catalog data — re-queries fresh data and repopulates the cache.
 */
async function syncCatalog() {
  'use server'
  after(async () => {
    await syncCatalogFromUpstream()
    // Purge the \"products\" cache so the next read (in this view or any
    // other view sharing the tag) re-queries fresh data. The \"max\"
    // profile keeps the repopulated cache long-lived until the next
    // sync, preserving the fast read-heavy browsing path.
    revalidateTag('products', 'max')
  })
}

let lastSyncedAt = null

/**
 * Raw catalog query. This is the expensive call we want to avoid
 * repeating on every request, so it is wrapped in `unstable_cache`
 * (see lib/products.js) and tagged \"products\".
 */
export async function getAllProducts() {
  // Simulate a database round-trip.
  await new Promise((resolve) => setTimeout(resolve, 100))
  return products.map((p) => ({ ...p }))
}

export function getLastSyncedAt() {
  return lastSyncedAt
}

/**
 * Simulate pulling fresh pricing / inventory / availability from the
 * upstream ERP/PIM. In production this can touch thousands of SKUs, so
 * it must never block the admin request — it is run in the background
 * via `after()` by the sync Server Action.
 *
 * Returns a summary of what was synced.
 */
export async function syncCatalogFromUpstream() {
  // Simulate a slow upstream sync: network fetch + bulk writes.
  await new Promise((resolve) => setTimeout(resolve, 2000))

  const syncedAt = Date.now()
  products = products.map((p) => {
    // Simulate upstream drift: inventory moves, availability flips,
    // and occasional price changes.
    const drift = Math.floor((Math.random() - 0.5) * 20)
    const inventory = Math.max(0, p.inventory + drift)
    return {
      ...p,
      inventory,
      available: inventory > 0,
      price: Math.random() > 0.8 ? Math.round(p.price * 1.05) : p.price,
      syncedAt,
    }
  })
  lastSyncedAt = syncedAt
  return { count: products.length, syncedAt }
}

import { getAllProducts, getLastSyncedAt } from './db'

// Read-heavy catalog access: cache the product query and tag it
// \"products\" so every catalog view shares a single cache entry, and a
// single `revalidateTag(\"products\")` refreshes them all. The data is
// cached indefinitely and only re-queried when the \"products\" tag is
// invalidated (by the sync action), which keeps day-to-day browsing fast.
export const getProducts = unstable_cache(
  async () => {
    const items = await getAllProducts()
    return { items, lastSyncedAt: getLastSyncedAt() }
  },
  ['products'],
  { tags: ['products'] }
)
