// Simulated product catalog backed by an upstream ERP/PIM.
//
// `getAllProducts` is the read path used by catalog views. It is wrapped in
// `unstable_cache` (tagged \"products\") at the call site so day-to-day browsing
// is served from cache without re-querying the data source on every request.
//
// `syncCatalogFromERP` simulates pulling the latest pricing, inventory, and
// availability from the upstream ERP/PIM. In production this would hit the
// PIM API and write updated rows to the database; here it mutates the
// in-memory source so a subsequent cache refill observes fresh data.

let products = [
  { id: 1, name: 'Laptop', price: 999, stock: 32, available: true },
  { id: 2, name: 'Phone', price: 699, stock: 54, available: true },
  { id: 3, name: 'Tablet', price: 499, stock: 18, available: true },
  { id: 4, name: 'Headphones', price: 199, stock: 0, available: false },
  { id: 5, name: 'Watch', price: 299, stock: 27, available: true },
]

export async function getAllProducts() {
  // Simulate database / PIM read latency.
  await new Promise((resolve) => setTimeout(resolve, 100))

  // Return a defensive copy so callers cannot mutate the source of truth.
  return products.map((p) => ({ ...p }))
}

// Simulate a sync job that pulls the latest catalog from the upstream ERP/PIM.
// In practice this can touch thousands of SKUs, so it is intentionally run
// out of the request path (via `after` in the Server Action) and the catalog
// cache is invalidated with stale-while-revalidate semantics so admins keep
// working while the refresh settles.
export async function syncCatalogFromERP() {
  // Simulate a long-running upstream sync (pricing, inventory, availability).
  await new Promise((resolve) => setTimeout(resolve, 1500))

  // Apply simulated upstream changes: price drift, restocks, availability flips.
  products = products.map((p) => ({
    ...p,
    // Small price drift to mimic ERP repricing.
    price: p.price + Math.round((Math.random() - 0.5) * 20),
    // Restock a portion of inventory.
    stock: Math.max(0, p.stock + Math.round((Math.random() - 0.3) * 40)),
    // Flip availability when stock runs out.
    available: p.stock > 0,
  }))

  return { syncedAt: new Date().toISOString(), skuCount: products.length }
}
