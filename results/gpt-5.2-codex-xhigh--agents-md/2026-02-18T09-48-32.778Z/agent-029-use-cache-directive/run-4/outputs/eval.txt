function resolveAvailability(inventory: number, status: Product['status']): Product['availability'] {
  if (status === 'Archived') {
    return 'Discontinued';
  }
  if (inventory === 0) {
    return 'Backorder';
  }
  if (inventory < 20) {
    return 'Low stock';
  }
  return 'In stock';
}

export async function requestCatalogSync(): Promise<void> {
  await new Promise(resolve => setTimeout(resolve, 60));
}

export async function getCatalogSnapshot(): Promise<CatalogSnapshot> {
  'use cache';
  cacheLife('max');
  cacheTag('products');

  const now = new Date();
  const products = PRODUCT_SEEDS.map(seed => {
    const updatedAt = new Date(now.getTime() - seed.updatedHoursAgo * 60 * 60 * 1000);
    return {
      ...seed,
      updatedAt: updatedAt.toISOString(),
      availability: resolveAvailability(seed.inventory, seed.status)
    };
  });

  const lowStock = products.filter(product => product.availability === 'Low stock').length;
  const outOfStock = products.filter(product => product.inventory === 0).length;
  const paused = products.filter(product => product.status !== 'Active').length;
  const activeChannels = new Set(products.map(product => product.channel)).size;

  return {
    refreshedAt: now.toISOString(),
    products,
    stats: {
      totalSkus: products.length,
      lowStock,
      outOfStock,
      paused,
      activeChannels
    }
  };
}
