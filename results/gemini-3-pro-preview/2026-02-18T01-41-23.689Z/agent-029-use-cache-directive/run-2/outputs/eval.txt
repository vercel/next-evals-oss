// Use /tmp for writable access in serverless environments
const DB_PATH = path.join('/tmp', 'products.json');

type Product = {
  id: string;
  name: string;
  price: number;
  stock: number;
};

// Initialize mock database if not exists
function ensureDB() {
  if (!fs.existsSync(DB_PATH)) {
    const initialProducts: Product[] = [
      { id: '1', name: 'Product A', price: 100, stock: 50 },
      { id: '2', name: 'Product B', price: 200, stock: 20 },
      { id: '3', name: 'Product C', price: 300, stock: 0 },
    ];
    fs.writeFileSync(DB_PATH, JSON.stringify(initialProducts));
  }
}

// Helper to read/write (synchronous for simplicity in mock)
function readDB(): Product[] {
  ensureDB();
  try {
    return JSON.parse(fs.readFileSync(DB_PATH, 'utf-8'));
  } catch (e) {
    return [];
  }
}

function writeDB(data: Product[]) {
  ensureDB();
  fs.writeFileSync(DB_PATH, JSON.stringify(data));
}

// Simulate slow external sync
export async function syncProducts() {
    // Simulate slow upstream fetch
    await new Promise((resolve) => setTimeout(resolve, 2000));
    
    // Read current data
    let currentProducts = readDB();

}
