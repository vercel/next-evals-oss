import fs from 'fs/promises';
import path from 'path';

const DB_PATH = path.join(process.cwd(), 'products-db.json');

export async function getDB() {
  try {
    const data = await fs.readFile(DB_PATH, 'utf8');
    return JSON.parse(data);
  } catch (error) {
    // Initialize with mock data if file doesn't exist
    const initial = {
      lastSynced: new Date().toISOString(),
      products: Array.from({ length: 20 }, (_, i) => ({
        id: `sku-${1000 + i}`,
        name: `Product Item ${i + 1}`,
        price: (Math.random() * 100 + 10).toFixed(2),
        inventory: Math.floor(Math.random() * 200),
        status: Math.random() > 0.2 ? 'Active' : 'Draft'
      }))
    };
    await fs.writeFile(DB_PATH, JSON.stringify(initial, null, 2));
    return initial;
  }
}

export async function updateDB() {
  // Simulate fetching fresh data and updating our local \"database\"
  const db = await getDB();
  
  db.lastSynced = new Date().toISOString();
  // Randomize inventory to demonstrate updates
  db.products = db.products.map((p: any) => ({
    ...p,
    inventory: Math.floor(Math.random() * 200),
    price: (Math.random() * 100 + 10).toFixed(2)
  }));
  
  await fs.writeFile(DB_PATH, JSON.stringify(db, null, 2));
}
