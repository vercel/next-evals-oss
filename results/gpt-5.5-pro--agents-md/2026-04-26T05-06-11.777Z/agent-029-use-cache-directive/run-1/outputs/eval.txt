  await new Promise((resolve) => setTimeout(resolve, 100))

  return baseProducts.map((product, index) => {
    const inventoryDelta = catalogRevision * (index % 2 === 0 ? 3 : -2)
    const inventory = Math.max(0, product.inventory + inventoryDelta)
    const price = product.price + catalogRevision * (index % 3 === 0 ? 8 : 5)

    return {
      ...product,
      price,
      inventory,
      availability: formatAvailability(inventory),
      updatedAt: lastSyncedAt.toISOString(),
    }
  })
}

  // Simulate a larger upstream job without making the form submission wait.

  catalogRevision += 1
  lastSyncedAt = new Date()

  return {
    revision: catalogRevision,
    syncedAt: lastSyncedAt.toISOString(),
  }
}
