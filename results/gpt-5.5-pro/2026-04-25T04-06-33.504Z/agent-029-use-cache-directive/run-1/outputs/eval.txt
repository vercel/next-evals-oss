export type Product = {
  id: number
  sku: string
  name: string

  price: number
  inventory: number
  availability: 'In stock' | 'Low stock' | 'Unavailable'
  updatedAt: string
}

export type ProductCatalog = {
  products: Product[]
  refreshedAt: string
  totalInventory: number
  unavailableCount: number
}

async function loadProductCatalog(): Promise<ProductCatalog> {
  const products = (await getAllProducts()) as Product[]

  return {
    products,
    refreshedAt: new Date().toISOString(),
    totalInventory: products.reduce((total, product) => total + product.inventory, 0),
    unavailableCount: products.filter((product) => product.availability === 'Unavailable').length,
  }
}
