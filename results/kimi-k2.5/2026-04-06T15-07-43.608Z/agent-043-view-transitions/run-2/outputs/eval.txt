'use client'

import Link from 'next/link'
import { useRouter } from 'next/navigation'
import { useCallback, useRef } from 'react'

interface ViewTransitionLinkProps {
  href: string
  children: React.ReactNode
  className?: string
  direction?: 'forward' | 'back'
}

export function ViewTransitionLink({ 
  href, 
  children, 
  className,
  direction = 'forward'
}: ViewTransitionLinkProps) {
  const router = useRouter()
  const linkRef = useRef<HTMLAnchorElement>(null)

  const handleClick = useCallback((e: React.MouseEvent<HTMLAnchorElement>) => {
    
    const prefersReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches
    
    
    if (prefersReducedMotion || !document.startViewTransition) {
      return 
    }

    e.preventDefault()

    
    sessionStorage.setItem('view-transition-direction', direction)

    
    document.startViewTransition(() => {
      router.push(href)
    })
  }, [href, router, direction])

  return (
    <Link 
      ref={linkRef}
      href={href} 
      className={className}
      onClick={handleClick}
    >
      {children}
    </Link>
  )
}


export function useTransitionDirection() {
  if (typeof window === 'undefined') return 'forward'
  
  const direction = sessionStorage.getItem('view-transition-direction') || 'forward'
  
  sessionStorage.removeItem('view-transition-direction')
  return direction as 'forward' | 'back'
}

'use client'

import { useEffect, useState, useCallback } from 'react'
import { usePathname, useRouter } from 'next/navigation'


const PRODUCT_IMAGE_PREFIX = 'product-image-'

export function getProductViewTransitionName(slug: string): string {
  return `${PRODUCT_IMAGE_PREFIX}${slug}`
}


export function useViewTransitionNavigation() {
  const router = useRouter()
  const pathname = usePathname()
  const [isTransitioning, setIsTransitioning] = useState(false)
  const [direction, setDirection] = useState<'forward' | 'back'>('forward')

  const navigate = useCallback(async (
    href: string,
    options?: { 
      viewTransitionName?: string 
      direction?: 'forward' | 'back'
    }
  ) => {
    const navDirection = options?.direction || (
      href.split('/').length > pathname.split('/').length ? 'forward' : 'back'
    )
    
    setDirection(navDirection)

    
    const prefersReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches
    
    if (prefersReducedMotion || !document.startViewTransition) {
      router.push(href)
      return
    }

    setIsTransitioning(true)

    try {
      const transition = document.startViewTransition(() => {
        router.push(href)
      })

      await transition.ready
      
      
      await transition.finished
    } catch (error) {
      
      router.push(href)
    } finally {
      setIsTransitioning(false)
    }
  }, [router, pathname])

  return { navigate, isTransitioning, direction }
}


export function usePageTransition(direction: 'forward' | 'back') {
  useEffect(() => {
    const prefersReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches
    if (prefersReducedMotion || !document.startViewTransition) return

    const pageElement = document.querySelector('[data-page-container]')
    if (!pageElement) return

    
    pageElement.setAttribute('data-transition-direction', direction)

    return () => {
      pageElement.removeAttribute('data-transition-direction')
    }
  }, [direction])
}


export function PageTransitionContainer({ 
  children,
  direction = 'forward'
}: { 
  children: React.ReactNode
  direction?: 'forward' | 'back'
}) {
  return (
    <div 
      data-page-container 
      data-transition-direction={direction}
      className=\"page-transition-container\"
    >
      {children}
    </div>
  )
}

'use client'

import Link from 'next/link'
import { useRouter } from 'next/navigation'
import { useCallback, useRef } from 'react'

interface ViewTransitionLinkProps {
  href: string
  children: React.ReactNode
  className?: string
  direction?: 'forward' | 'back'
}

export function ViewTransitionLink({ 
  href, 
  children, 
  className,
  direction = 'forward'
}: ViewTransitionLinkProps) {
  const router = useRouter()
  const linkRef = useRef<HTMLAnchorElement>(null)

  const handleClick = useCallback((e: React.MouseEvent<HTMLAnchorElement>) => {
    
    const prefersReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches
    
    
    if (prefersReducedMotion || !document.startViewTransition) {
      return 
    }

    e.preventDefault()

    
    sessionStorage.setItem('view-transition-direction', direction)

    
    document.startViewTransition(() => {
      router.push(href)
    })
  }, [href, router, direction])

  return (
    <Link 
      ref={linkRef}
      href={href} 
      className={className}
      onClick={handleClick}
    >
      {children}
    </Link>
  )
}


export function useTransitionDirection() {
  if (typeof window === 'undefined') return 'forward'
  
  const direction = sessionStorage.getItem('view-transition-direction') || 'forward'
  
  sessionStorage.removeItem('view-transition-direction')
  return direction as 'forward' | 'back'
}
