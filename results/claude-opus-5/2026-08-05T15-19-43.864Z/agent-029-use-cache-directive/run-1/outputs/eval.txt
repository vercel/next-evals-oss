import { getProducts } from '@/lib/products'
import { requestCatalogSync } from '@/lib/catalog-sync'

export default async function Page() {
  // Served from the `products` cache entry, so day-to-day browsing never
  // re-queries the database.
  const products = await getProducts()

  async function syncCatalog() {
    'use server'

    // Kick off the upstream ERP/PIM sync. Returns as soon as the job is queued.
    await requestCatalogSync()

    // Expire the shared `products` tag with a cacheLife profile (\"max\") rather
    // than calling `updateTag('products')`. Passing a profile marks the entry
    // stale-while-revalidate: this request returns immediately instead of
    // blocking while the catalog is recomputed, and Next refreshes the entry in
    // the background so every view reading `getProducts()` converges shortly
    // after. `updateTag` would instead force read-your-own-writes and make the
    // admin wait on a freshly recomputed catalog.
    revalidateTag('products', 'max')
  }

  return (
    <main>
      <h1>Product catalog</h1>

      <form action={syncCatalog}>
        <button type=\"submit\">Sync latest catalog</button>
      </form>

      <table>
        <thead>
          <tr>
            <th>ID</th>
            <th>Name</th>
            <th>Price</th>
          </tr>
        </thead>

            </tr>
          ))}
        </tbody>
      </table>
    </main>
  )
}
