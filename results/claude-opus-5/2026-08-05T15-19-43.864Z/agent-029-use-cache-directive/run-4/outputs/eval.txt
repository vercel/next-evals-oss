/**
 * Pull the latest pricing, inventory and availability from the upstream
 * ERP/PIM and persist it.
 *
 * A real sync touches thousands of SKUs, so treat this as the slow write: it is
 * why the admin page must not block on a freshly recomputed catalog.
 */
export async function syncCatalogFromUpstream() {
  // Simulate the upstream round-trip plus the bulk write.
  await new Promise((resolve) => setTimeout(resolve, 150))

  revision += 1
  syncedAt = new Date().toISOString()
  catalog = BASE_CATALOG.map((product, index) => ({
    ...product,
    // Deterministic stand-in for upstream deltas.
    price: product.price + ((revision * (index + 1) * 7) % 25),
    stock: Math.max(0, product.stock - ((revision * (index + 2)) % 5)),
  }))

  return { revision, syncedAt, skuCount: catalog.length }
}

/**
 * The one cached read for catalog data.
 *
 * Tagged `\"products\"`, so a single `revalidateTag('products', 'max')` refreshes
 * every view built on this function. Any new view that needs catalog data
 * should call this rather than hitting `lib/db` directly — that is what keeps
 * invalidation consistent across the app.
 *
 * `cacheLife('max')` means entries are not time-expired: the catalog is
 * read-heavy and changes only when a sync says so, so browsing never pays for a
 * re-query.
 */
export async function getCatalog(): Promise<Catalog> {
  'use cache'
  cacheTag('products')
  cacheLife('max')

  const [products, { revision, syncedAt }] = await Promise.all([
    getAllProducts(),
    getCatalogRevision(),
  ])

  return { products, revision, syncedAt }
}
