import { PRODUCTS_TAG, getProducts, syncCatalogFromUpstream } from '@/lib/products'

export default async function ProductCatalogPage() {
  const { products, refreshedAt } = await getProducts()

  async function syncLatestCatalog() {
    'use server'

    // Pull the latest pricing / inventory / availability from the ERP/PIM.
    await syncCatalogFromUpstream()

    // Mark the catalog stale rather than expiring it. The next visit to any
    // `\"products\"`-tagged view is served the cached list immediately while the
    // fresh one is recomputed in the background (stale-while-revalidate), so
    // nobody waits on a recompute over thousands of SKUs.
    //
    // `updateTag(PRODUCTS_TAG)` — or the deprecated one-argument

    // the next read a blocking cache miss. That buys read-your-own-writes we
    // don't need here, at the cost of the responsiveness we do.
    revalidateTag(PRODUCTS_TAG, 'max')
  }

  return (

      <h1>Product catalog</h1>

import { cacheLife, cacheTag } from 'next/cache'

import { getAllProducts, pullUpstreamCatalog } from './db'

/** Cache tag shared by every view that reads catalog data. */
export const PRODUCTS_TAG = 'products'

/**
 * Cached read of the catalog. Browsing is read-heavy, so requests are served
 * from the cache instead of re-querying the database each time.
 *
 * Every view that calls this shares the `\"products\"` tag, so one invalidation
 * refreshes all of them.
 */
export async function getProducts() {
  'use cache'
  cacheTag(PRODUCTS_TAG)
  // Catalog freshness is driven by the sync action below, not by a timer, so
  // keep entries around for as long as possible and let the tag do the work.
  cacheLife('max')

  const products = await getAllProducts()

  return { products, refreshedAt: new Date().toISOString() }
}

/** Run the upstream sync. Called from a Server Action, never from a render. */
export async function syncCatalogFromUpstream() {
  return pullUpstreamCatalog()
}
