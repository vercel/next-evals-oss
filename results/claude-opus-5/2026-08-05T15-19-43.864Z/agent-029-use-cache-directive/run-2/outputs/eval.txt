/**
 * Pulls pricing / inventory / availability from the upstream ERP/PIM feed and
 * writes it back to the catalog. Real sync jobs touch thousands of SKUs, so this
 * is deliberately slow — it is the work we do not want admins waiting on.
 */
export async function syncCatalogFromUpstream() {
  await new Promise((resolve) => setTimeout(resolve, 1500))

  revision += 1
  syncedAt = new Date().toISOString()
  products = products.map((product) => ({
    ...product,
    // Stand-in for upstream deltas: prices drift, stock gets recounted.
    price: product.price + revision,
    stock: Math.max(0, product.stock - revision),
  }))

  return { revision, syncedAt }
}

import { getAllProducts, getCatalogRevision } from './db.js'

export const PRODUCTS_TAG = 'products'

export type Product = {
  id: number
  name: string
  price: number
  stock: number
}

export type Catalog = {
  products: Product[]
  revision: number
  syncedAt: string
}

/**
 * The single cached read for catalog data. Every view that needs products calls
 * this, so they all share one cache entry and one invalidation tag — refreshing
 * the tag updates all of them together.
 */
export async function getCatalog(): Promise<Catalog> {
  'use cache'
  // Invalidation is driven by the sync action, not by elapsed time, so hold the
  // entry for as long as the cache allows rather than expiring it on a timer.
  cacheLife('max')
  cacheTag(PRODUCTS_TAG)

  const [products, { revision, syncedAt }] = await Promise.all([
    getAllProducts(),
    getCatalogRevision(),
  ])

  return { products, revision, syncedAt }
}
