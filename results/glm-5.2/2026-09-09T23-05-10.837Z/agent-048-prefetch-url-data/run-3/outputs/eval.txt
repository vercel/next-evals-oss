'use client'

import { useState } from 'react'
import Link from 'next/link'
import { useRouter } from 'next/navigation'
import type { PreparedCategory } from '@/lib/data'

export function CategoryNav({ categories }: { categories: PreparedCategory[] }) {
  const router = useRouter()
  const [selected, setSelected] = useState<PreparedCategory | null>(null)

  function handleClick(event: React.MouseEvent<HTMLAnchorElement>, category: PreparedCategory) {
    event.preventDefault()
    setSelected(category)
    router.push(`/catalog?category=${category.slug}`)
  }

  return (
    <>
      <ul>
        {categories.map((category) => (
          <li key={category.slug}>
            <Link
              href={`/catalog?category=${category.slug}`}
              onClick={(event) => handleClick(event, category)}
            >
              {category.label}
            </Link>
          </li>
        ))}
      </ul>

      {selected ? (
        <section>
          <h2>{selected.heading}</h2>
          <ul>
            {selected.products.map((product) => (
              <li key={product}>{product}</li>
            ))}
          </ul>
        </section>
      ) : null}
    </>
  )
}

export type CategorySlug = 'featured' | 'sale'

export type Category = {
  slug: CategorySlug
  label: string
  heading: string
}

export type PreparedCategory = Category & {
  products: string[]
}

const productsByCategory: Record<CategorySlug, string[]> = {
  featured: ['Desk Lamp', 'Wool Blanket'],
  sale: ['Canvas Tote', 'Travel Mug'],
}

export const categories: Category[] = [
  { slug: 'featured', label: 'Featured products', heading: 'Featured products' },
  { slug: 'sale', label: 'Products on sale', heading: 'Products on sale' },
]

export function getCategory(slug: string | undefined): Category | undefined {
  return categories.find((category) => category.slug === slug)
}

export async function getProducts(category: string): Promise<string[]> {
  'use cache'
  await new Promise((resolve) => setTimeout(resolve, 800))
  return productsByCategory[category as CategorySlug] ?? []
}

export async function prepareCategory(slug: CategorySlug): Promise<PreparedCategory> {
  const category = categories.find((c) => c.slug === slug)!
  const products = await getProducts(slug)
  return { ...category, products }
}

export async function prepareAllCategories(): Promise<PreparedCategory[]> {
  return Promise.all(categories.map((category) => prepareCategory(category.slug)))
}
