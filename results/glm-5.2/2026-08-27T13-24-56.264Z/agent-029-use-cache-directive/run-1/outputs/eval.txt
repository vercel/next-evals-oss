// Simulate pulling the latest pricing / inventory / availability from the
// upstream ERP/PIM and writing it into the local catalog. Real syncs touch
// thousands of SKUs, so this is the slow job we keep off the request path and
// run in the background instead of blocking the admin on it.
export async function syncProductsFromUpstream() {
  await new Promise((resolve) => setTimeout(resolve, 2000))

  const syncedAt = Date.now()
  store = store.map((product) => {
    const stock = Math.max(0, product.stock - 1)
    return {
      ...product,
      stock,
      status: stock === 0 ? 'backorder' : 'active',
      syncedAt,
    }
  })
  lastSyncedAt = syncedAt

  return store.length
}
