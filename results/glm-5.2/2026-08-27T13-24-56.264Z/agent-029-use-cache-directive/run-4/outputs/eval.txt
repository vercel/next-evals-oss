// Cache the catalog read under the shared \"products\" tag.
//
// - keyParts: ['products'] -> one warm cache entry shared by every catalog view.
// - tags: ['products']      -> a single revalidateTag('products') invalidates it
//                              (and every other consumer of the same tag) at once.
// - revalidate: false       -> stay warm until an admin sync explicitly
//                              invalidates it, so day-to-day browsing never
//                              re-queries the catalog.
export const getCatalog = unstable_cache(
  async (): Promise<Catalog> => {
    const products = (await getAllProducts()) as Product[]
    return {
      products,
      revision: products[0]?.revision ?? 0,
      syncedAt: products[0]?.syncedAt ?? '',
      fetchedAt: new Date().toISOString(),
    }
  },
  ['products'],
  { tags: ['products'], revalidate: false }
)
