'use client'

import { useSearchParams } from 'next/navigation'
import { useEffect, useSyncExternalStore } from 'react'
import { getCachedCategory, prefetchCategory, subscribeToCategory } from '@/lib/catalog-cache'

export function CatalogContent() {
  const searchParams = useSearchParams()
  const category = searchParams.get('category') ?? 'featured'

  const data = useSyncExternalStore(
    subscribeToCategory,
    () => getCachedCategory(category),
    () => undefined
  )

  useEffect(() => {
    prefetchCategory(category)
  }, [category])

  if (!data) {
    return <p>Loading category...</p>
  }

  return (
    <section>
      <h2>{data.heading}</h2>
      <ul>
        {data.products.map((product) => (
          <li key={product}>{product}</li>
        ))}
      </ul>
    </section>
  )
}

'use client'

import Link from 'next/link'
import { useEffect } from 'react'
import { prefetchCategory } from '@/lib/catalog-cache'
import type { Category } from '@/lib/data'

export function CategoryNav({ categories }: { categories: Category[] }) {
  useEffect(() => {
    categories.forEach((category) => prefetchCategory(category.slug))
  }, [categories])

  return (
    <ul>
      {categories.map((category) => (

            onMouseEnter={() => prefetchCategory(category.slug)}
            onFocus={() => prefetchCategory(category.slug)}
          >
            {category.label}
          </Link>
        </li>
      ))}
    </ul>
  )
}

export type CategoryData = {
  heading: string
  products: string[]
}

const cache = new Map<string, CategoryData>()
const pending = new Map<string, Promise<CategoryData>>()
const listeners = new Set<() => void>()

function notify() {
  for (const listener of listeners) listener()
}

async function fetchCategoryData(category: string): Promise<CategoryData> {
  const response = await fetch(`/api/catalog?category=${encodeURIComponent(category)}`)
  return response.json()
}



export function prefetchCategory(category: string) {
  if (cache.has(category) || pending.has(category)) return

  const promise = fetchCategoryData(category)
    .then((data) => {
      cache.set(category, data)
      notify()
      return data
    })
    .finally(() => pending.delete(category))

  pending.set(category, promise)
}

export function getCachedCategory(category: string): CategoryData | undefined {
  return cache.get(category)
}

export function subscribeToCategory(listener: () => void): () => void {
  listeners.add(listener)
  return () => listeners.delete(listener)
}

export type Category = { slug: string; label: string }

export const categories: Category[] = [
  { slug: 'featured', label: 'Featured products' },
  { slug: 'sale', label: 'Products on sale' },
]

const productsByCategory: Record<string, string[]> = {
  featured: ['Desk Lamp', 'Wool Blanket'],
  sale: ['Canvas Tote', 'Travel Mug'],
}

export function getCategoryHeading(category: string) {
  return categories.find((c) => c.slug === category)?.label ?? categories[0].label
}

export async function getProducts(category: string) {
  await new Promise((resolve) => setTimeout(resolve, 800))
  return productsByCategory[category] ?? []
}

export async function getCategoryData(category: string) {
  return {
    heading: getCategoryHeading(category),
    products: await getProducts(category),
  }
}
