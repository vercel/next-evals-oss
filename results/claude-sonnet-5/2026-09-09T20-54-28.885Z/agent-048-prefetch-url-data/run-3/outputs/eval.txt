'use client'

import { useEffect, useSyncExternalStore } from 'react'

export type CatalogData = {
  heading: string
  products: string[]
}

const resolved = new Map<string, CatalogData>()
const pending = new Map<string, Promise<CatalogData>>()
const listeners = new Set<() => void>()

function notify() {
  for (const listener of listeners) listener()
}

export function prefetchCatalog(category: string) {
  if (resolved.has(category) || pending.has(category)) return

  const promise = fetch(`/api/catalog?category=${encodeURIComponent(category)}`)
    .then((res) => res.json() as Promise<CatalogData>)
    .then((data) => {
      resolved.set(category, data)
      pending.delete(category)
      notify()
      return data
    })

  pending.set(category, promise)
}

function subscribe(listener: () => void) {
  listeners.add(listener)
  return () => listeners.delete(listener)
}

export function useCatalogData(category: string): CatalogData | null {
  useEffect(() => {
    prefetchCatalog(category)
  }, [category])

  return useSyncExternalStore(
    subscribe,
    () => resolved.get(category) ?? null,
    () => null,
  )
}
