}

import { getProducts } from '@/lib/data'

export type CategoryEntry = {
  heading: string
  products: string[]
}

const cache = new Map<string, CategoryEntry>()
const pending = new Map<string, Promise<CategoryEntry>>()
const listeners = new Set<() => void>()

function headingFor(category: string) {
  return category === 'sale' ? 'Products on sale' : 'Featured products'
}

function load(category: string): Promise<CategoryEntry> {
  const inflight = pending.get(category)
  if (inflight) return inflight

  const promise = getProducts(category).then((products) => {
    const entry = { heading: headingFor(category), products }
    cache.set(category, entry)
    pending.delete(category)
    listeners.forEach((listener) => listener())
    return entry
  })

  pending.set(category, promise)
  return promise
}

export function prefetchCategory(category: string) {
  if (cache.has(category)) return
  void load(category)
}

export function ensureCategory(category: string) {
  return cache.get(category) ? Promise.resolve(cache.get(category)!) : load(category)
}

export function getCachedCategory(category: string) {
  return cache.get(category)
}

export function subscribe(listener: () => void) {
  listeners.add(listener)
  return () => listeners.delete(listener)
}
