type PendingNavigation = {

function prefersReducedMotion() {
  return window.matchMedia('(prefers-reduced-motion: reduce)').matches
}

export function ViewTransitionProvider({ children }: { children: ReactNode }) {
  const router = useRouter()
  const pathname = usePathname()
  const pendingNavigation = useRef<PendingNavigation | null>(null)

  useEffect(() => {
    const pending = pendingNavigation.current
    if (!pending || pending.pathname !== pathname) return

    clearTimeout(pending.timeout)
    pendingNavigation.current = null
    pending.resolve()
  }, [pathname])

  useEffect(
    () => () => {
      const pending = pendingNavigation.current
      if (!pending) return
      clearTimeout(pending.timeout)
      pending.resolve()
      pendingNavigation.current = null
    },
    [],
  )

  const navigate = useCallback(
    (href: string, direction: Direction) => {
      const destination = new URL(href, window.location.href)
      if (destination.origin !== window.location.origin || destination.pathname === pathname) {
        router.push(href)
        return
      }

      if (!document.startViewTransition || prefersReducedMotion()) {
        router.push(href)
        return
      }

      document.documentElement.dataset.navigationDirection = direction
      const transition = document.startViewTransition(
        () =>
          new Promise<void>((resolve) => {
            const timeout = setTimeout(() => {
              pendingNavigation.current = null
              resolve()
            }, 3000)
            pendingNavigation.current = { pathname: destination.pathname, resolve, timeout }
            router.push(href)
          }),
      )

      void transition.finished.then(
        () => delete document.documentElement.dataset.navigationDirection,
        () => delete document.documentElement.dataset.navigationDirection,
      )
    },
    [pathname, router],
  )

  return (
    <NavigationContext.Provider value={{ navigate }}>
      {children}
    </NavigationContext.Provider>
  )
}

export function TransitionLink({
  href,
  direction = 'forward',
  onClick,
  target,
  download,
  ...props

async function ProductDetails({ slug }: { slug: string }) {
  await new Promise((resolve) => setTimeout(resolve, 100))
  const product = getProduct(slug)
  if (!product) notFound()

  return (
    <div className=\"product-info content-reveal\">

  params,
}: {
  params: Promise<{ slug: string }>
}) {
  const { slug } = await params
  const product = products.find((p) => p.slug === slug)
  if (!product) notFound()

  return (
    <div>

          }}
        />
        <Suspense fallback={<DetailsSkeleton />}>
          <ProductDetails slug={slug} />
        </Suspense>
      </div>
    </div>
  )
}

import { notFound } from 'next/navigation'
import { TransitionLink } from '@/app/ViewTransitionProvider'

async function ProductDetails({ slug }: { slug: string }) {
  await new Promise((resolve) => setTimeout(resolve, 100))
  const product = getProduct(slug)
  if (!product) notFound()

  return (
    <div className=\"product-info content-reveal\">
      <h1>{product.name}</h1>
      <p className=\"product-price\">${product.price}</p>
      <p className=\"product-description\">{product.description}</p>
    </div>
  )
}

function DetailsSkeleton() {
  return (
    <div className=\"product-info\">
      <div className=\"skeleton-text large\" />
      <div className=\"skeleton-text short\" />
      <div className=\"skeleton-text\" />
    </div>
  )
}

export default async function ProductPage({
  params,
}: {
  params: Promise<{ slug: string }>
}) {
  const { slug } = await params
  const product = products.find((p) => p.slug === slug)
  if (!product) notFound()

  return (
    <div>
      <TransitionLink href=\"/\" direction=\"back\" className=\"back-link\">
        ← Back to products
      </TransitionLink>
      <div className=\"product-detail\">
        <div
          className=\"product-hero\"
          style={{
            backgroundColor: product.color,
            viewTransitionName: `product-image-${product.slug}`,
          }}
        />
        <Suspense fallback={<DetailsSkeleton />}>
          <ProductDetails slug={slug} />
        </Suspense>
      </div>
    </div>
  )
}

'use client'

import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useRef,
  type ComponentProps,
  type MouseEvent,
  type ReactNode,
} from 'react'
import Link from 'next/link'
import { usePathname, useRouter } from 'next/navigation'

type Direction = 'forward' | 'back'
type NavigationContextValue = {
  navigate: (href: string, direction: Direction) => void
}

const NavigationContext = createContext<NavigationContextValue | null>(null)

type PendingNavigation = {
  pathname: string
  resolve: () => void
  timeout: ReturnType<typeof setTimeout>
}

function prefersReducedMotion() {
  return window.matchMedia('(prefers-reduced-motion: reduce)').matches
}

export function ViewTransitionProvider({ children }: { children: ReactNode }) {
  const router = useRouter()
  const pathname = usePathname()
  const pendingNavigation = useRef<PendingNavigation | null>(null)

  useEffect(() => {
    const pending = pendingNavigation.current
    if (!pending || pending.pathname !== pathname) return

    clearTimeout(pending.timeout)
    pendingNavigation.current = null
    pending.resolve()
  }, [pathname])

  useEffect(
    () => () => {
      const pending = pendingNavigation.current
      if (!pending) return
      clearTimeout(pending.timeout)
      pending.resolve()
      pendingNavigation.current = null
    },
    [],
  )

  const navigate = useCallback(
    (href: string, direction: Direction) => {
      const destination = new URL(href, window.location.href)
      if (destination.origin !== window.location.origin || destination.pathname === pathname) {
        router.push(href)
        return
      }

      if (!document.startViewTransition || prefersReducedMotion()) {
        router.push(href)
        return
      }

      document.documentElement.dataset.navigationDirection = direction
      const transition = document.startViewTransition(
        () =>
          new Promise<void>((resolve) => {
            const timeout = setTimeout(() => {
              pendingNavigation.current = null
              resolve()
            }, 3000)
            pendingNavigation.current = { pathname: destination.pathname, resolve, timeout }
            router.push(href)
          }),
      )

      void transition.finished.then(
        () => delete document.documentElement.dataset.navigationDirection,
        () => delete document.documentElement.dataset.navigationDirection,
      )
    },
    [pathname, router],
  )

  return (
    <NavigationContext.Provider value={{ navigate }}>
      {children}
    </NavigationContext.Provider>
  )
}

type TransitionLinkProps = Omit<ComponentProps<typeof Link>, 'href' | 'onClick'> & {
  href: string
  direction?: Direction
  onClick?: (event: MouseEvent<HTMLAnchorElement>) => void
}

export function TransitionLink({
  href,
  direction = 'forward',
  onClick,
  target,
  download,
  ...props
}: TransitionLinkProps) {
  const context = useContext(NavigationContext)

  function handleClick(event: MouseEvent<HTMLAnchorElement>) {
    onClick?.(event)
    if (
      !context ||
      event.defaultPrevented ||
      event.button !== 0 ||
      event.metaKey ||
      event.ctrlKey ||
      event.shiftKey ||
      event.altKey ||
      (target !== undefined && target !== '_self') ||
      download !== undefined
    ) {
      return
    }

    event.preventDefault()
    context.navigate(href, direction)
  }

  return (
    <Link
      href={href}
      target={target}
      download={download}
      onClick={handleClick}
      {...props}
    />
  )
}
