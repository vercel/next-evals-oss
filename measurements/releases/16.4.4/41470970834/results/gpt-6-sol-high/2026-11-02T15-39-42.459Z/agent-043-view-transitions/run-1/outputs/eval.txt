import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useRef,
  type ReactNode,
} from 'react'

type Direction = 'forward' | 'back'
type NavigationContextValue = (href: string, direction: Direction) => void

const NavigationContext = createContext<NavigationContextValue | null>(null)

export function NavigationTransitions({ children }: { children: ReactNode }) {
  const router = useRouter()
  const pathname = usePathname()
  const pending = useRef<{ target: string; finish: () => void } | null>(null)
  const transitioning = useRef(false)

  
  
  
  useEffect(() => {
    if (pending.current?.target === pathname) pending.current.finish()
  }, [pathname])

  useEffect(() => {
    return () => pending.current?.finish()
  }, [])

  const navigate = useCallback<NavigationContextValue>(
    (href, direction) => {
      if (transitioning.current) {
        router.push(href)
        return
      }

      transitioning.current = true
      document.documentElement.dataset.navigationDirection = direction

      const transition = document.startViewTransition(() =>
        new Promise<void>((resolve) => {
          const timeout = window.setTimeout(finish, 5000)

          function finish() {
            window.clearTimeout(timeout)
            pending.current = null
            resolve()
          }

          pending.current = { target: href, finish }
          router.push(href)
        }),
      )

      const cleanup = () => {
        pending.current?.finish()
        transitioning.current = false
        delete document.documentElement.dataset.navigationDirection
      }
      void transition.finished.then(cleanup, cleanup)
    },
    [router],
  )

  return (
    <NavigationContext.Provider value={navigate}>
      {children}
    </NavigationContext.Provider>
  )
}

export function TransitionLink({
  href,
  direction,
  children,
  ...props

  direction: Direction

      {children}
    </Link>
  )

import { Suspense } from 'react'
import { getProduct, products } from '@/lib/products'
import { notFound } from 'next/navigation'
import { TransitionLink } from '@/app/NavigationTransitions'

async function ProductDetails({ slug }: { slug: string }) {
  await new Promise((resolve) => setTimeout(resolve, 100))
  const product = getProduct(slug)
  if (!product) notFound()

  return (
    <div className=\"product-info suspense-reveal\">
      <div className=\"suspense-reveal-skeleton\" aria-hidden=\"true\">
        <DetailsSkeleton />
      </div>
      <div className=\"suspense-reveal-content\">
        <h1>{product.name}</h1>

        <p className=\"product-description\">{product.description}</p>
      </div>
    </div>
  )
}

  return (
    <div className=\"product-info\">
      <div className=\"skeleton-text large\" />
      <div className=\"skeleton-text short\" />
      <div className=\"skeleton-text\" />
    </div>
  )
}

export default async function ProductPage({
  params,
}: {
  params: Promise<{ slug: string }>
}) {
  const { slug } = await params
  const product = products.find((p) => p.slug === slug)
  if (!product) notFound()

  return (
    <main>
      <TransitionLink href=\"/\" direction=\"back\" className=\"back-link\">
        ← Back to products
      </TransitionLink>
      <div className=\"product-detail\">
        <div
          className=\"product-hero\"
          style={{
            backgroundColor: product.color,
            viewTransitionName: `product-${product.slug}`,
          }}
        />
        <Suspense fallback={<DetailsSkeleton />}>
          <ProductDetails slug={slug} />
        </Suspense>
      </div>
    </main>
  )
}

import Link from 'next/link'
import { usePathname, useRouter } from 'next/navigation'
import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useRef,
  type ReactNode,
} from 'react'

type Direction = 'forward' | 'back'
type NavigationContextValue = (href: string, direction: Direction) => void

const NavigationContext = createContext<NavigationContextValue | null>(null)

export function NavigationTransitions({ children }: { children: ReactNode }) {
  const router = useRouter()
  const pathname = usePathname()
  const pending = useRef<{ target: string; finish: () => void } | null>(null)
  const transitioning = useRef(false)

  
  
  
  useEffect(() => {
    if (pending.current?.target === pathname) pending.current.finish()
  }, [pathname])

  useEffect(() => {
    return () => pending.current?.finish()
  }, [])

  const navigate = useCallback<NavigationContextValue>(
    (href, direction) => {
      if (transitioning.current) {
        router.push(href)
        return
      }

      transitioning.current = true
      document.documentElement.dataset.navigationDirection = direction

      const transition = document.startViewTransition(() =>
        new Promise<void>((resolve) => {
          const timeout = window.setTimeout(finish, 5000)

          function finish() {
            window.clearTimeout(timeout)
            pending.current = null
            resolve()
          }

          pending.current = { target: href, finish }
          router.push(href)
        }),
      )

      const cleanup = () => {
        pending.current?.finish()
        transitioning.current = false
        delete document.documentElement.dataset.navigationDirection
      }
      void transition.finished.then(cleanup, cleanup)
    },
    [router],
  )

  return (
    <NavigationContext.Provider value={navigate}>
      {children}
    </NavigationContext.Provider>
  )
}

export function TransitionLink({
  href,
  direction,
  children,
  ...props
}: Omit<React.ComponentProps<typeof Link>, 'href' | 'onNavigate'> & {
  href: string
  direction: Direction
}) {
  const navigate = useContext(NavigationContext)

  return (
    <Link
      href={href}
      prefetch={true}
      {...props}
      onNavigate={(event) => {
        if (
          navigate &&
          typeof document.startViewTransition === 'function' &&
          !window.matchMedia('(prefers-reduced-motion: reduce)').matches
        ) {
          event.preventDefault()
          navigate(href, direction)
        }
      }}
    >
      {children}
    </Link>
  )
}
