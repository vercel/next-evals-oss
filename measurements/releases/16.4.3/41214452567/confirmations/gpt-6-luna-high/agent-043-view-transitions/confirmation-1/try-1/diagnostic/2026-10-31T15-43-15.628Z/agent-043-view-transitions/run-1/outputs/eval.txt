import { Suspense } from 'react'
import { getProduct, products } from '@/lib/products'
import { notFound } from 'next/navigation'
import { TransitionLink } from '@/app/NavigationTransitions'

async function ProductDetails({ slug }: { slug: string }) {
  await new Promise((resolve) => setTimeout(resolve, 100))
  const product = getProduct(slug)
  if (!product) notFound()

  return (
    <div className=\"product-info\">
      <h1>{product.name}</h1>
      <p className=\"product-price\">${product.price}</p>
      <p className=\"product-description\">{product.description}</p>
    </div>
  )
}

  return (
    <div className=\"product-info\">

      <div className=\"skeleton-text short\" />
      <div className=\"skeleton-text\" />
    </div>
  )
}

export default async function ProductPage({
  params,
}: {
  params: Promise<{ slug: string }>
}) {
  const { slug } = await params
  const product = products.find((p) => p.slug === slug)
  if (!product) notFound()

  return (
    <div>
      <TransitionLink href=\"/\" className=\"back-link\">
        ← Back to products
      </TransitionLink>
      <div className=\"product-detail\">
        <div
          className=\"product-hero\"
          style={{
            backgroundColor: product.color,
            viewTransitionName: `product-image-${product.slug}`,
          }}
        />
        <Suspense fallback={<DetailsSkeleton />}>
          <ProductDetails slug={slug} />
        </Suspense>
      </div>
    </div>
  )
}

    finished: Promise<void>
  }

  pathname: string
  resolve: () => void

  navigate: (href: string) => void
}

const NavigationContext = createContext<NavigationContextValue | null>(null)

export function NavigationTransitions({ children }: { children: ReactNode }) {
  const router = useRouter()
  const pathname = usePathname()
  const pendingNavigation = useRef<PendingNavigation | null>(null)
  const transitionId = useRef(0)

  useEffect(() => {
    if (pendingNavigation.current?.pathname === pathname) {
      pendingNavigation.current.resolve()
      pendingNavigation.current = null
    }
  }, [pathname])

  const navigate = useCallback(
    (href: string) => {
      const destination = new URL(href, window.location.href)
      const viewTransitionDocument = document as ViewTransitionDocument
      const startViewTransition = viewTransitionDocument.startViewTransition
      const reducedMotion = window.matchMedia(
        '(prefers-reduced-motion: reduce)',
      ).matches

      if (
        !startViewTransition ||
        reducedMotion ||
        destination.origin !== window.location.origin
      ) {
        router.push(href)
        return
      }

      pendingNavigation.current?.resolve()
      const thisTransition = ++transitionId.current
      const direction = destination.pathname === '/' ? 'back' : 'forward'
      document.documentElement.dataset.transitionDirection = direction

      const transition = startViewTransition.call(
        viewTransitionDocument,
        () =>
          new Promise<void>((resolve) => {
            let settled = false
            const finish = () => {
              if (settled) return
              settled = true
              window.clearTimeout(timeout)
              if (pendingNavigation.current?.resolve === finish) {
                pendingNavigation.current = null
              }
              resolve()
            }
            const timeout = window.setTimeout(finish, 1800)

            pendingNavigation.current = {
              pathname: destination.pathname,
              resolve: finish,
            }
            router.push(href)

            
            if (destination.pathname === pathname) {
              window.requestAnimationFrame(finish)
            }
          }),
      )

      const clearDirection = () => {
        if (transitionId.current === thisTransition) {
          delete document.documentElement.dataset.transitionDirection
        }
      }
      void transition.finished.then(clearDirection, clearDirection)
    },
    [pathname, router],
  )

  return (
    <NavigationContext.Provider value={{ navigate }}>
      {children}
    </NavigationContext.Provider>
  )
}

type TransitionLinkProps = {
  href: string
  children: ReactNode
  className?: string
}

export function TransitionLink({
  href,
  children,
  className,
}: TransitionLinkProps) {
  const navigation = useContext(NavigationContext)

  if (!navigation) {
    throw new Error('TransitionLink must be rendered inside NavigationTransitions')
  }

  function handleClick(event: MouseEvent<HTMLAnchorElement>) {
    if (
      event.defaultPrevented ||
      event.button !== 0 ||
      event.metaKey ||
      event.ctrlKey ||
      event.shiftKey ||
      event.altKey
    ) {
      return
    }

    event.preventDefault()
    navigation.navigate(href)
  }

  return (
    <Link href={href} className={className} onClick={handleClick}>
      {children}
    </Link>
  )
}
