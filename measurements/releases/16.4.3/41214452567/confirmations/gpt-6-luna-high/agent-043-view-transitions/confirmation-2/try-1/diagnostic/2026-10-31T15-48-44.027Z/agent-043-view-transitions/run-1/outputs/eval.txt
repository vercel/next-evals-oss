'use client'

import {
  createContext,
  useCallback,
  useContext,
  useLayoutEffect,
  useRef,
} from 'react'
import Link from 'next/link'
import { usePathname, useRouter } from 'next/navigation'

type Direction = 'forward' | 'backward'
type TransitionOptions = {
  update: () => void | Promise<void>
  types?: string[]
}
type BrowserViewTransition = {
  finished: Promise<void>
}
type ViewTransitionDocument = Document & {
  startViewTransition?: (
    options: TransitionOptions | (() => void | Promise<void>),
  ) => BrowserViewTransition
}

type NavigateWithTransition = (href: string, direction: Direction) => void

const TransitionContext = createContext<NavigateWithTransition | null>(null)

export function ViewTransitions({ children }: { children: React.ReactNode }) {
  const router = useRouter()
  const pathname = usePathname()
  const pendingRoute = useRef<{
    pathname: string
    finish: () => void
  } | null>(null)

  useLayoutEffect(() => {
    const pending = pendingRoute.current
    if (pending && pending.pathname === pathname) pending.finish()
  }, [pathname])

  const navigate = useCallback<NavigateWithTransition>(
    (href, direction) => {
      const targetPath = new URL(href, window.location.href).pathname
      if (targetPath === pathname) {
        router.push(href)
        return
      }

      const documentWithTransitions = document as ViewTransitionDocument
      const startViewTransition = documentWithTransitions.startViewTransition
      const reducedMotion = window.matchMedia(
        '(prefers-reduced-motion: reduce)',
      ).matches

      if (reducedMotion) {
        router.push(href)
        return
      }

      if (!startViewTransition) {
        document.documentElement.dataset.navigationDirection = direction
        document.documentElement.dataset.cssNavigationFallback = 'true'
        router.push(href)
        window.setTimeout(() => {
          delete document.documentElement.dataset.navigationDirection
          delete document.documentElement.dataset.cssNavigationFallback
        }, 360)
        return
      }

      document.documentElement.dataset.navigationDirection = direction
      let timeout = 0
      let pending: { pathname: string; finish: () => void } | null = null
      const finish = () => {
        window.clearTimeout(timeout)
        if (pendingRoute.current === pending) pendingRoute.current = null
      }

      try {
        const transition = startViewTransition.call(documentWithTransitions, {
          types: [direction],
          update: () =>
            new Promise<void>((resolve) => {
              pending = {
                pathname: targetPath,
                finish: () => {
                  finish()
                  resolve()
                },
              }
              pendingRoute.current = pending
              timeout = window.setTimeout(() => {
                finish()
                resolve()
              }, 1500)
              router.push(href)
            }),
        })
        const clearDirection = () => {
          delete document.documentElement.dataset.navigationDirection
        }
        void transition.finished.then(clearDirection, clearDirection)
      } catch {
        finish()
        delete document.documentElement.dataset.navigationDirection
        router.push(href)
      }
    },
    [pathname, router],
  )

  return (
    <TransitionContext.Provider value={navigate}>
      {children}
    </TransitionContext.Provider>
  )
}

export function TransitionLink({
  href,
  direction,
  className,
  children,
}: {
  href: string
  direction: Direction
  className?: string
  children: React.ReactNode
}) {
  const navigate = useContext(TransitionContext)
  if (!navigate) throw new Error('TransitionLink must be inside ViewTransitions')

  return (
    <Link
      href={href}
      className={className}
      onNavigate={(event) => {
        event.preventDefault()
        navigate(href, direction)
      }}
    >
      {children}
    </Link>
  )
}

'use client'

import {
  createContext,
  useCallback,
  useContext,
  useLayoutEffect,
  useRef,
} from 'react'
import Link from 'next/link'
import { usePathname, useRouter } from 'next/navigation'

type Direction = 'forward' | 'backward'
type TransitionOptions = {
  update: () => void | Promise<void>
  types?: string[]
}
type BrowserViewTransition = {
  finished: Promise<void>
}
type ViewTransitionDocument = Document & {
  startViewTransition?: (
    options: TransitionOptions | (() => void | Promise<void>),
  ) => BrowserViewTransition
}

type NavigateWithTransition = (href: string, direction: Direction) => void

const TransitionContext = createContext<NavigateWithTransition | null>(null)

export function ViewTransitions({ children }: { children: React.ReactNode }) {
  const router = useRouter()
  const pathname = usePathname()
  const pendingRoute = useRef<{
    pathname: string
    finish: () => void
  } | null>(null)

  useLayoutEffect(() => {
    const pending = pendingRoute.current
    if (pending && pending.pathname === pathname) pending.finish()
  }, [pathname])

  const navigate = useCallback<NavigateWithTransition>(
    (href, direction) => {
      const targetPath = new URL(href, window.location.href).pathname
      if (targetPath === pathname) {
        router.push(href)
        return
      }

      const documentWithTransitions = document as ViewTransitionDocument
      const startViewTransition = documentWithTransitions.startViewTransition
      const reducedMotion = window.matchMedia(
        '(prefers-reduced-motion: reduce)',
      ).matches

      if (reducedMotion) {
        router.push(href)
        return
      }

      if (!startViewTransition) {
        document.documentElement.dataset.navigationDirection = direction
        document.documentElement.dataset.cssNavigationFallback = 'true'
        router.push(href)
        window.setTimeout(() => {
          delete document.documentElement.dataset.navigationDirection
          delete document.documentElement.dataset.cssNavigationFallback
        }, 360)
        return
      }

      document.documentElement.dataset.navigationDirection = direction
      let timeout = 0
      let pending: { pathname: string; finish: () => void } | null = null
      const finish = () => {
        window.clearTimeout(timeout)
        if (pendingRoute.current === pending) pendingRoute.current = null
      }

      try {
        const transition = startViewTransition.call(documentWithTransitions, {
          types: [direction],
          update: () =>
            new Promise<void>((resolve) => {
              pending = {
                pathname: targetPath,
                finish: () => {
                  finish()
                  resolve()
                },
              }
              pendingRoute.current = pending
              timeout = window.setTimeout(() => {
                finish()
                resolve()
              }, 1500)
              router.push(href)
            }),
        })
        const clearDirection = () => {
          delete document.documentElement.dataset.navigationDirection
        }
        void transition.finished.then(clearDirection, clearDirection)
      } catch {
        finish()
        delete document.documentElement.dataset.navigationDirection
        router.push(href)
      }
    },
    [pathname, router],
  )

  return (
    <TransitionContext.Provider value={navigate}>
      {children}
    </TransitionContext.Provider>
  )
}

export function TransitionLink({
  href,
  direction,
  className,
  children,
}: {
  href: string
  direction: Direction
  className?: string
  children: React.ReactNode
}) {
  const navigate = useContext(TransitionContext)
  if (!navigate) throw new Error('TransitionLink must be inside ViewTransitions')

  return (
    <Link
      href={href}
      className={className}
      onNavigate={(event) => {
        event.preventDefault()
        navigate(href, direction)
      }}
    >
      {children}
    </Link>
  )
}
