import {
  createContext,
  useCallback,
  useContext,
  useLayoutEffect,
  useRef,
  type ReactNode,
} from 'react'
import { usePathname, useRouter } from 'next/navigation'

type ViewTransitionHandle = {
  finished: Promise<void>
}

type ViewTransitionDocument = Document & {
  startViewTransition?: (
    callback: () => void | Promise<void>,
  ) => ViewTransitionHandle
}

type PendingNavigation = {
  pathname: string
  resolve: () => void
}

type ViewTransitionContextValue = {
  navigate: (href: string, direction: 'forward' | 'back') => void
}

const ViewTransitionContext = createContext<ViewTransitionContextValue | null>(
  null,
)

export function ViewTransitionProvider({
  children,
}: {
  children: ReactNode
}) {
  const router = useRouter()
  const pathname = usePathname()
  const currentPathname = useRef(pathname)
  const pendingNavigation = useRef<PendingNavigation | null>(null)

  useLayoutEffect(() => {
    currentPathname.current = pathname
    if (pendingNavigation.current?.pathname === pathname) {
      pendingNavigation.current.resolve()
      pendingNavigation.current = null
    }
  }, [pathname])

  const navigate = useCallback(
    (href: string, direction: 'forward' | 'back') => {
      const targetPathname = new URL(href, window.location.href).pathname
      const viewTransitionDocument = document as ViewTransitionDocument
      const reducedMotion = window.matchMedia(
        '(prefers-reduced-motion: reduce)',
      ).matches

      if (
        reducedMotion ||
        !viewTransitionDocument.startViewTransition ||
        targetPathname === currentPathname.current
      ) {
        router.push(href)
        return
      }

      
      
      pendingNavigation.current?.resolve()

      document.documentElement.dataset.navigationDirection = direction
      let resolvePathChange: () => void = () => {}
      const pathChange = new Promise<void>((resolve) => {
        resolvePathChange = resolve
        pendingNavigation.current = {
          pathname: targetPathname,
          resolve,
        }
      })

      const transition = viewTransitionDocument.startViewTransition(
        async () => {
          router.push(href)
          
          
          await Promise.race([
            pathChange,
            new Promise<void>((resolve) => window.setTimeout(resolve, 4000)),
          ])
        },
      )

      const cleanup = () => {
        if (pendingNavigation.current?.pathname === targetPathname) {
          pendingNavigation.current = null
          resolvePathChange()
        }
        delete document.documentElement.dataset.navigationDirection
      }

      void transition.finished.then(cleanup, cleanup)
    },
    [router],
  )

  return (
    <ViewTransitionContext.Provider value={{ navigate }}>
      {children}
    </ViewTransitionContext.Provider>
  )
}

export function useViewTransitionNavigation() {
  const context = useContext(ViewTransitionContext)
  if (!context) {
    throw new Error(
      'useViewTransitionNavigation must be used inside ViewTransitionProvider',
    )
  }
  return context.navigate
}

import { Suspense } from 'react'
import { getProduct, products } from '@/lib/products'
import { notFound } from 'next/navigation'
import { TransitionLink } from '../../TransitionLink'

async function ProductDetails({ slug }: { slug: string }) {
  await new Promise((resolve) => setTimeout(resolve, 100))
  const product = getProduct(slug)
  if (!product) notFound()

  return (
    <div className=\"product-info suspense-content\">
      <h1>{product.name}</h1>
      <p className=\"product-price\">${product.price}</p>
      <p className=\"product-description\">{product.description}</p>
    </div>
  )
}

function DetailsSkeleton() {
  return (
    <div className=\"product-info suspense-skeleton\">
      <div className=\"skeleton-text large\" />
      <div className=\"skeleton-text short\" />
      <div className=\"skeleton-text\" />
    </div>
  )
}

export default async function ProductPage({
  params,
}: {
  params: Promise<{ slug: string }>
}) {
  const { slug } = await params
  const product = products.find((p) => p.slug === slug)
  if (!product) notFound()

  return (
    <div>
      <TransitionLink href=\"/\" direction=\"back\" className=\"back-link\">
        ← Back to products
      </TransitionLink>
      <div className=\"product-detail\">
        <div
          className=\"product-hero\"
          style={{
            backgroundColor: product.color,
            viewTransitionName: `product-${product.slug}`,
          }}
        />
        <Suspense fallback={<DetailsSkeleton />}>
          <ProductDetails slug={slug} />
        </Suspense>
      </div>
    </div>
  )
}

'use client'

import Link from 'next/link'
import type { ReactNode } from 'react'
import { useViewTransitionNavigation } from './ViewTransitionProvider'

export function TransitionLink({
  href,
  direction,
  className,
  children,
}: {
  href: string
  direction: 'forward' | 'back'
  className?: string
  children: ReactNode
}) {
  const navigate = useViewTransitionNavigation()

  return (
    <Link
      href={href}
      className={className}
      onNavigate={(event) => {
        event.preventDefault()
        navigate(href, direction)
      }}
    >
      {children}
    </Link>
  )
}

'use client'

import {
  createContext,
  useCallback,
  useContext,
  useLayoutEffect,
  useRef,
  type ReactNode,
} from 'react'
import { usePathname, useRouter } from 'next/navigation'

type ViewTransitionHandle = {
  finished: Promise<void>
}

type ViewTransitionDocument = Document & {
  startViewTransition?: (
    callback: () => void | Promise<void>,
  ) => ViewTransitionHandle
}

type PendingNavigation = {
  pathname: string
  resolve: () => void
}

type ViewTransitionContextValue = {
  navigate: (href: string, direction: 'forward' | 'back') => void
}

const ViewTransitionContext = createContext<ViewTransitionContextValue | null>(
  null,
)

export function ViewTransitionProvider({
  children,
}: {
  children: ReactNode
}) {
  const router = useRouter()
  const pathname = usePathname()
  const currentPathname = useRef(pathname)
  const pendingNavigation = useRef<PendingNavigation | null>(null)

  useLayoutEffect(() => {
    currentPathname.current = pathname
    if (pendingNavigation.current?.pathname === pathname) {
      pendingNavigation.current.resolve()
      pendingNavigation.current = null
    }
  }, [pathname])

  const navigate = useCallback(
    (href: string, direction: 'forward' | 'back') => {
      const targetPathname = new URL(href, window.location.href).pathname
      const viewTransitionDocument = document as ViewTransitionDocument
      const reducedMotion = window.matchMedia(
        '(prefers-reduced-motion: reduce)',
      ).matches

      if (
        reducedMotion ||
        !viewTransitionDocument.startViewTransition ||
        targetPathname === currentPathname.current
      ) {
        router.push(href)
        return
      }

      
      
      pendingNavigation.current?.resolve()

      document.documentElement.dataset.navigationDirection = direction
      let resolvePathChange: () => void = () => {}
      const pathChange = new Promise<void>((resolve) => {
        resolvePathChange = resolve
        pendingNavigation.current = {
          pathname: targetPathname,
          resolve,
        }
      })

      const transition = viewTransitionDocument.startViewTransition(
        async () => {
          router.push(href)
          
          
          await Promise.race([
            pathChange,
            new Promise<void>((resolve) => window.setTimeout(resolve, 4000)),
          ])
        },
      )

      const cleanup = () => {
        if (pendingNavigation.current?.pathname === targetPathname) {
          pendingNavigation.current = null
          resolvePathChange()
        }
        delete document.documentElement.dataset.navigationDirection
      }

      void transition.finished.then(cleanup, cleanup)
    },
    [router],
  )

  return (
    <ViewTransitionContext.Provider value={{ navigate }}>
      {children}
    </ViewTransitionContext.Provider>
  )
}

export function useViewTransitionNavigation() {
  const context = useContext(ViewTransitionContext)
  if (!context) {
    throw new Error(
      'useViewTransitionNavigation must be used inside ViewTransitionProvider',
    )
  }
  return context.navigate
}
