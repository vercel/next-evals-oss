export const categories = [
  { slug: 'featured', label: 'Featured products', heading: 'Featured products' },
  { slug: 'sale', label: 'Products on sale', heading: 'Products on sale' },
]

const productsByCategory: Record<string, string[]> = {
  featured: ['Desk Lamp', 'Wool Blanket'],
  sale: ['Canvas Tote', 'Travel Mug'],
}

export function getProducts(category: string) {
  return productsByCategory[category] ?? []
}

export function getCategoryHeading(category: string) {
  return categories.find((item) => item.slug === category)?.heading ?? 'Featured products'
}
